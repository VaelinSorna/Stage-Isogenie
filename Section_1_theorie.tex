\section{Rappels théoriques et notations}

\subsection{Théorie des nombres}
%pas/peu de preuves
 Le stage a débuté par une mise à niveau en théorie des nombres, dont on énonce ici les résultats principaux utiles par la suite. On se réfère à \cite{Cox}[Ch 5, 7]. On ne s'intéresse qu'à certaines extensions de degrés $2$ de $\Q$.

\subsubsection{Ordres quadratiques imaginaires}

\begin{Def}
	
	Un corps $\K$ est appelé corps quadratique imaginaire si $\K = \Q(\sqrt{D})$ avec $D$ entier négatif sans facteur carré. Un ordre de $\K$ est un sous anneaux unitaire de $\K$, finement engendré comme $\Z$-module et contenant une $\Q$-base de $\K$. Un tel anneaux est appelé un ordre quadratique imaginaire. 
\end{Def}

\begin{Prop}[\cite{Cox}(Prop 5.3, 7.2)]
	
	Soit $\OR$ un ordre quadratique imaginaire de $\K$. Alors $\OR$ est un $\Z$-module libre de rang $2$. De plus $\K = Frac(\OR)$.
	On note $\OR_{\K}$ son anneau des entiers algébriques. C'est un ordre de $\K$, maximal pour l'inclusion et contenant tout les ordres de $\K$. 
\end{Prop}

\begin{Def}
	Le discriminant d'un corps quadratique imaginaire $\K = \Q(\sqrt{D})$, noté $d_{\K}$, est défini par :
	\begin{enumerate}
		\item $d_{\K} = D$ si $D = 1 [4]$,
		\item $d_{\K} = 4D$ sinon.
	\end{enumerate}
	En particulier $d_{\K} = 0 ,\;1[4]$. On pose de plus $w_{\K} = \frac{d_{\K} + \sqrt{d_{\K}}}{2}$
	
\end{Def}


\begin{Prop}[\cite{Cox}(Prop 5.14, 7.2)]\label{baseOrdreWk}
	
	Soit $\K$ un corps quadratique imaginaire. Alors $\OR_{\K} = \Z\left[ w_{\K}\right] $. De plus
	\begin{enumerate}
		\item Si $d_\K \neq 1 [4]$, $\OR_{\K} = \Z\left[ \sqrt{d_\K}\right] $
		\item Sinon, $\OR_{\K} = \Z\left[ \frac{1 + \sqrt{d_\K}}{2}\right] $
	\end{enumerate} 
	Soit $\OR$ un ordre de $\K$ un corps et $f = \left[ \OR_{\K} : \OR\right]$. Alors $\OR = \Z\left[ fw_{\K}\right] $.  
\end{Prop}

\begin{Def}
	Lorsque $\K$ est un corps quadratique imaginaire, si $\OR$ est un ordre de $\K$, $f = \left[ \OR_{\K} : \OR\right]$ est appelé le conducteur de $\OR$, et $\Delta = f^2d_{\K}$ son discriminant. $\OR$ est alors noté $\OR_{\Delta}$.
\end{Def}

\begin{Prop}
	Soit $\OR$, $\OR'$ deux ordres d'un $\K$ quadratique imaginaire. Soit $f$ et $f'$ leurs conducteurs. Si $\OR'\subset\OR$ alors $f$ divise $f'$
\end{Prop}

\begin{Def}
	Un réseau $\Lambda$ de $\C$ est un ensemble de la forme $\left\lbrace nz_{1} + mz_{2} , n,m\in\Z\right\rbrace$ où $z_{1},\; z_{2}\in\C$ sont linéairement indépendant sur $\R$. On note alors $\Lambda = \left[ z_{1}, z_{2}\right] $
\end{Def}

\begin{Rem}
	Les ordres quadratiques imaginaires sont donc des réseaux de $\C$, qui contiennent $1$ : $\OR_{\K} = \left[1 , w_{\K}\right]$, et si $\OR$ est l'ordre de conducteur $f$, alors $\OR = \left[ 1, fw_{\K}\right]$.
\end{Rem}

Les ordres quadratiques imaginaires ne sont pas des anneaux factoriels en général. On va cependant obtenir des résultats de factorisations au niveau des idéaux. 

\begin{Prop}[\cite{Cox}(Prop 5.4, Ex 5.1, Ex 7.4)]
	
	 Soit $\LR\subset\OR$ un idéal d'un ordre quadratique imaginaire.
	\begin{enumerate}
		\item Il existe $a\in\Z$ appartenant à $\LR$
		\item $\LR$ est un $\Z$-module libre de rang $2$
		\item Le quotient $\OR/\LR$ est fini.
		\item Si $\LR$ est premier, alors $\LR$ est maximal. 
	\end{enumerate}
\end{Prop}

\begin{Def}
	La norme d'un idéal $\LR\subset\OR$ d'un ordre d'un corps quadratique imaginaire $\K$ est $N(\LR) = |\OR/\LR|$.
\end{Def}

\begin{Coro}\label{formeIdeauxBaseWk}
	Soit $\OR$ un ordre de $\K$ un corps quadratique imaginaire de conducteur $f$. Tout idéal $\LR$ de $\OR$ est un réseaux de $\C$ de la forme $\left[ a, b + cfw_{\K}\right]$ avec $a | N(\LR)$, $b$, $c\in\Z$, $c\neq 0$ et $0 \leq b< a$. 	
\end{Coro}

\begin{The}[\cite{Cox}(Prop 5.5,Cor 5.6)]
	
	$\OR_{\K}$ est un anneaux de Dedekind, c'est à dire :
	\begin{enumerate}
		\item $\OR_{\K}$ est intégralement clos
		\item $\OR_{\K}$ est noethérien
		\item Ses idéaux premier sont maximaux 
	\end{enumerate}
	En particulier tout idéal $\LR$ admet une unique factorisation par des idéaux premier (à permutation prêt des facteurs)
	\begin{equation*}
		\LR = \PR_{1}^{e_{1}}\PR_{2}^{e_{2}}\ldots\PR_{g}^{e_{g}}
	\end{equation*}
	avec $e_{1}$, ... $e_{g}\in\N$ et $g\in\N$.
\end{The}

Un ordre $\OR\subset\K$ quelconque n'est pas un anneaux de Dedekind, mais seule la clôture intégrale pose problème.

\begin{Prop}[\cite{Cox}(5.A, Ex 7.4)]
	
	Un ordre $\OR$ d'un corps de nombre $\K$ est noethérien, et ses idéaux premiers sont maximaux.
\end{Prop}

\subsubsection{Idéaux fractionnaires et groupes de classes d'idéaux}

\begin{Def}
	Soit $\K$ un corps quadratique imaginaire et $\OR\subset\K$ un ordre. On définit une relation d'équivalence entre idéaux $\LR,\;\LR'\subset\OR$.
	\begin{equation*}
		\LR\simeq\LR'\iff\LR = \alpha\LR', \alpha\in\K.
	\end{equation*} 
	Un idéal fractionnaire de $\OR$ est un $\OR$-module de $\K$ de type fini, non nul. Les idéaux de $\OR$ seront appelé "idéaux entiers" pour les distinguer des idéaux fractionnaires. 
	On note $I(\OR_{\K})$ l'ensemble des idéaux fractionnaires de $\OR_{\K}$.
\end{Def}

\begin{Prop}[\cite{Cox}(5.A, Ex 5.2)]
	
	Les idéaux fractionnaires d'un ordre quadratique imaginaire $\OR$ sont les ensembles de la forme $\alpha\LR$, où $\alpha\in\K$ et $\LR\subset\OR$ est un idéal entier de $\OR$.
\end{Prop}

\begin{Prop}[\cite{Cox}(Prop 5.7)]
	 
	Soit $\K$ un corps quadratique imaginaire. Les idéaux fractionnaires de $\OR_{\K}$ sont inversibles, c'est à dire que pour tout $\IR\in I(\OR_{\K})$, il existe $\IR'\in I(\OR_{\K})$ tel que $(\IR)(\IR') = \OR_{\K}$.
	$I(\OR_{\K})$ est donc un groupe pour la loi multiplicative. 
	Tout idéal fractionnaire $\IR$ admet une unique factorisation par des idéaux entiers premiers ou leurs inverses (à permutation prêt des facteurs):
	\begin{equation*}
		\IR = \PR_{1}^{e_{1}}\PR_{2}^{e_{2}}\ldots\PR_{g}^{e_{g}}
	\end{equation*}
	avec $e_{1}$, ... $e_{g}\in\Z$ et $g\in\N$.
\end{Prop}

\begin{Prop}[\cite{Cox}(Prop 5.7)]\label{denomid}
	
	Soit $\IR$ un idéal fractionnaire d'un ordre quadratique imaginaire $\OR$. Il existe un entier $d$ tel que $\LR = d\IR$ soit un idéal entier. 
\end{Prop}


\begin{Def}
	Pour tout corps de nombre $\K$, $I(\OR_{\K})$ contient l'ensemble $P(\OR_{\K})$ des idéaux fractionnaires principaux, i.e de la forme $\alpha\OR_{\K}$ avec $\alpha\in\K$. 
\end{Def} 

\begin{Rem}
	L'inverse de $\alpha\OR_{\K}$ dans $I(\OR_{\K})$ est $\alpha^{-1}\OR_{\K}$ : $P(\OR_{\K}))$ est un sous-groupe de $I(\OR_{\K})$. Le quotient $I(\OR_{\K}/P(\OR_{\K}))$ pour la loi multiplicative correspond aux classes d'équivalences de la relation $\IR\simeq\IR'\iff\IR = \alpha\IR',\alpha\in\\K$.
\end{Rem}

\begin{Def}
	Pour tout corps quadratique imaginaire $\K$, on appelle groupe de classes d'idéaux de $\OR_{\K}$ (où de $\K$) le quotient $Cl(\OR_{\K}) =  I(\OR_{\K})/P(\OR_{\K})$. Si le contexte le permet, on appellera $Cl(\OR_{\K})$ le groupe de classes de $\K$. 
\end{Def}


La notion de groupe de classes peut être défini pour tout anneaux de Dedekind, en considérant le corps de fraction. Une propriété remarquable de $\OR_{\K}$ comparé aux anneaux de Dedekind quelconques est que $Cl(\OR_{\K})$ est fini. 

\begin{Prop}[\cite{Cox}(5.A)]
	
	$Cl(\OR_{\K})$ est fini, de cardinal $h(\OR_{\K})$. Alors $h(\OR_{\K}) = O(\sqrt{|d_\K})$
\end{Prop} 


On va maintenant définir une notion groupe de classes pour un ordre $\OR$ quelconque d'un corps quadratique imaginaire $\K$. La difficulté vient du fait que certains idéaux entiers d'un ordre $\OR$ ne seront pas inversibles. Si $\LR\subset\OR$ est un idéal, $\LR$ pourrait aussi être un idéal d'un ordre contenant $\OR$. On ne pourra définir une notion d'inverse pour $\LR$ qu'en le voyant comme idéal du plus gros anneaux possible, c'est à dire $\left\lbrace \alpha\in\K, \alpha\LR\subset\LR\right\rbrace$. 

\begin{Def}
	Soit $\OR$ un ordre d'un corps quadratique imaginaire $\K$. $\LR\subset\OR$ est un idéal propre de $\OR$ si c'est un idéal entier tel que
	\begin{equation*}
		\OR = \left\lbrace \alpha\in\K, \alpha\LR\subset\LR\right\rbrace
	\end{equation*} 
	$\IR$ est un idéal fractionnaire propre de $\OR$ s'il vérifie $\OR = \left\lbrace \alpha\in\K, \alpha\IR\subset\IR\right\rbrace$. On note $I(\OR)$ l'ensemble des idéaux fractionnaires propres.
\end{Def}

\begin{Prop}[\cite{Cox}(7.4)]
	
	Un idéal fractionnaire $\IR$ d'un ordre $\OR$ est inversible si et seulement s'il est propre.
\end{Prop}

\begin{Coro}
	Les idéaux fractionnaires principaux d'un ordre $\OR$ sont propres, et forment un sous groupe de $I(\OR)$.
\end{Coro}

\begin{Def}
	Soit $\OR$ un ordre quadratique imaginaire. On note $P(\OR)$ le sous groupe de $I(\OR)$ des idéaux fractionnaires principaux. Le groupe de classes d'idéaux de $\OR$ est le quotient $Cl(\OR) =  I(\OR)/P(\OR)$
\end{Def}

\begin{Prop}[\cite{Cox}(7.A)]\label{cardinalGroupeClasse}
	
	$Cl(\OR)$ est fini, on note $h(\OR)$ son cardinal. Alors $h(\OR) = O(\sqrt{|\Delta|})$
\end{Prop} 



\begin{Prop}[\cite{Cox}(Prop 7.14)]\label{NormesId}
	
	Soit $\OR$ un ordre d'un corps quadratique imaginaire $\K$, et $\LR$, $\LR'\in I(\OR)$. On note $\bar{\LR}$ l'idéal conjugué de $\LR$ selon la conjugaison dans $\K$. 
	\begin{enumerate}
		\item $\forall\alpha\in\OR,\; N(\alpha\OR) = N_{\Q}^{\K}(\alpha)$
		\item $N(\LR\LR') = N(\LR)N(\LR')$ pour tout idéal entier propre $\LR'$.
		\item $\LR\bar{\LR} = N(\LR)\OR$, donc $\bar{\LR}$ est l'inverse de $\LR$ dans $Cl(\OR)$
		\item $\LR$ est un idéal principal si et seulement s'il existe un élément de $\LR$ de norme égale à $N(\LR)$.
	\end{enumerate}
\end{Prop}


Dans un ordre quadratique imaginaire $\OR$, on ne pourra factoriser de façon unique que les idéaux entiers de norme premier avec le conducteur. 

\begin{Def}[\cite{Cox}(Lem 7.18)]
	Soit $\LR\subset\OR$ un idéal entier d'un ordre quadratique imaginaire, et $f$ le conducteur de $\OR$. L'idéal $\LR$ est dit premier avec le conducteur si l'une des deux conditions équivalente suivante est vérifiée :
	\begin{enumerate}
		\item $N(\LR)$ est premier avec $f$
		\item $\LR + f\OR = \OR$
	\end{enumerate}
\end{Def}

\begin{Prop}[\cite{Cox}(Lem 7.18)]
	
	Si $\LR\subset\OR$ est un idéal premier avec le conducteur, alors il est propre. L'ensemble des idéaux premier avec le conducteur est stable par multiplication et forme un sous groupe de $I(\OR)$.  
\end{Prop}

\begin{Def}
	On note $I(\OR,f)$ le sous groupe de $I(\OR)$ engendré par les idéaux premier avec le conducteur, et $P(\OR,f)$ le sous groupe de $I(\OR,f)$ engendré par les idéaux principaux premier avec le conducteur.
\end{Def}

\begin{Prop}[\cite{Cox}(Lem 7.19)]
	
	Si $\OR$ est un ordre d'un corps quadratique imaginaire, de conducteur $f$, alors
	\begin{equation*}
		Cl(\OR) =  I(\OR, f)/P(\OR, f)
	\end{equation*} 
\end{Prop}

\begin{Rem}
	La proposition précédente implique en particulier qu'il existe dans toute classes d'idéaux un représentant premier avec le conducteur. Si $\OR = \OR_{\K}$, $f = 1$ et aucune restriction n'est faite. 
\end{Rem}

\begin{Prop}[\cite{Cox}(7.C, Ex 7.26)]
	Un idéal $\LR\subset\OR$ premier avec le conducteur admet une unique factorisation (à permutation prêt des facteurs) par des idéaux premier, et premiers avec le conducteur.
	
	\begin{equation*}
		\LR = \PR_{1}^{e_{1}}\PR_{2}^{e_{2}}\ldots\PR_{g}^{e_{g}}
	\end{equation*}
	
	avec $e_{1}$, ... $e_{g}\in\N$ et $g\in\N$.
	
\end{Prop}




\subsubsection{Décomposition des nombres premiers}


Dans $\Z$, l'idéal engendré par un nombre premier $p$ est un idéal premier. Au contraire dans un ordre quadratique imaginaire $\OR$, l'idéal engendré $p\OR$ peut ne plus être premier. Cependant il se factorise lorsque $p$ est premier avec le conducteur. 


\begin{Def}
	Soit $\OR$ un ordre d'un corps quadratique imaginaire $\K$. Soit $p$ un nombre premier, premier avec le conducteur. Si un idéal premier $\PR$ apparaît dans la factorisation de $p\OR$, on dira que $\PR$ est au dessus de $p$. 
\end{Def}

Il est possible d'expliciter une factorisation de $p\OR$, d'après un théorème dû à Kummer. On énonce ce théorème dans le cas des corps quadratique imaginaire. Pour un énoncé général pour tout corps de nombre, on se réfère à \cite{Coh00}[Th 4.8.13].

\begin{The}[Théorème de Kummer, cas quadratique imaginaire]\label{Kummer}
	
	Soit $\OR_{\Delta}$ un ordre quadratique imaginaire d'un corps $\K$. Soit $T(X) = X^2 - \Delta \in\F_{p}\left[ X\right] $. Soit $p$ un nombre premier, premier avec le conducteur de $\OR_\Delta$
	\begin{enumerate}
		\item Si $T(X) = X^2$, alors 
		\begin{equation*}
			p\OR = \PR^{2}\; avec \; \PR = p\OR + \sqrt{\Delta}\OR.
		\end{equation*}
		
		\item Si $T(X)$ est irréductible, alors $p\OR$ est un idéal premier de norme $p^{2}$.
		
		\item Si $T(X) = T_{1}(X)T_{2}(X)$ se factorise de façon non trivial, alors 
		\begin{equation*}
			p\OR = \PR_{1}\PR_{2}\; avec \; \PR_{i} = p\OR + T_{i}(\sqrt{\Delta})\OR.
		\end{equation*}
		
	\end{enumerate}
\end{The}

\begin{Def}
	Soit $\OR_\Delta$ un ordre quadratique imaginaire. Soit un nombre premier $p$, premier avec le conducteur. Selon les trois cas possible du théorème précédent, on dira que $p$ est 
	\begin{enumerate}
		\item ramifié dans $\OR_\Delta$, si $\Delta = 0$ modulo $p$.
		\item inerte dans $\OR_\Delta$, si $\Delta$ n'est pas un carré modulo $p$.
		\item décomposé dans $\OR_\Delta$, si $\Delta$ est un carré modulo $p$.
	\end{enumerate}
\end{Def}

Ainsi connaître la factorisation de $p\OR_\Delta$ dans un ordre quadratique imaginaire $\OR_\Delta$ revient à calculer le symbole de Legendre $ \left( \frac{\Delta}{p}\right) $. Expliciter la factorisation dans le cas décomposé demande en plus de trouver les racines de $\Delta$ dans $\F_{p}$. 


\subsubsection{Corps de classe de Hilbert}

Le but de cette section est d'introduire une extension particulière d'un corps de nombres (quadratique imaginaire) $\K$, appelée le corps de classe de Hilbert de $\K$, dont le groupe de Galois est isomorphes à $Cl(\OR_{\K})$.

\begin{Def}
	Soit $\K$ un corps de nombres (quadratique imaginaire), et $\PR_{\K}\subset\OR_{\K}$ un idéal premier. Soit $\Lc/\K$ une extension galoisienne. L'idéal $\PR_{\K}\OR_{\Lc}$ se factorise dans $\OR_{\Lc}$ :
	\begin{equation*}
		\PR_{\K}\OR_{\Lc} = \PR_{1}^{e_{1}}\PR_{2}^{e_{2}}\ldots\PR_{g}^{e_{g}}
	\end{equation*} 
	On note $g$ le nombre d'idéaux premiers de la factorisation. Les exposants $e_{i}$ sont appelés les indices de ramification de $\PR_{\K}$ dans $\Lc$.
	Puisque $\PR_{\K}\OR_{\Lc}\subset\PR_{i}$ et que $\PR_{i}$ est maximal, le corps $\OR_{\Lc}/\PR_{i}$ est une extension du corps $\OR_{\K}/\PR_{\K}$ et on note $f_{i}$ son degré. Les degrés $f_{i}$ sont appelés les degrés d'inerties de $\PR_{\K}$ dans $\Lc$.
\end{Def}


\begin{Prop}[\cite{Cox}(Th 5.8, 5.9)]
	
	Soit $\K$ un corps de nombres, $\Lc$ une extension galoisienne de $\K$ et $n = \left[ \Lc : \K\right] $. Soit $\PR_{\K}\subset\OR_{\K}$ un idéal premier. Alors les coefficients $e_{i}$ et $f_{i}$ de $\PR_{\K}$ dans $\Lc$ ne dépendent pas de l'indice $i$ i.e. de l'idéal premier $\PR_{i}\subset\OR_{\Lc}$ correspondant.  
	On note $e$ et $f$ leur valeurs. Alors $efg = n$.
\end{Prop}

\begin{Def}
	Soit $\K$ un corps de nombres, $\Lc$ une extension galoisienne de $\K$. On note $n = \left[ \Lc : \K\right] $. Un idéal premier $\PR_{\K}\subset\OR_{\K}$ est dit :
	\begin{enumerate}
		\item ramifié dans $\Lc$, si son indice de ramification $e$ vérifie $e > 1$.
		\item inerte dans $\Lc$, si son degré d'inertie $f$ vérifie $f = n$, ce qui implique $e = g = 1$ et $p\OR_{\K}$ maximal.
		\item décomposé dans $\Lc$, si $g = n$, ce qui implique $e = f = 1$.
	\end{enumerate}
\end{Def}

\begin{Def} Soit $\Lc/\K$ une extension galoisienne d'un corps de nombre $\K$. Une injection $\sigma :\K\rightarrow\C$ est dite réelle si elle plonge $\K$ dans $\R$. Une injection réelle $\sigma$ de $\K$ est dite ramifiée dans $\Lc$ si elle peut se prolonger en une injection $\sigma :\Lc\rightarrow\C$ qui n'est pas réelle. 
\end{Def}

\begin{Rem}
	Les injections $\sigma :\K\rightarrow\C$ sont appelées les "premiers infinis" de $\K$. (voir \cite{Cox}(5.C))
\end{Rem}

\begin{Def}
	Une extension $\Lc/\K$ galoisienne d'un corps de nombre $\K$ est dit abélienne si son groupe de Galois est abéliens. Une telle extension est non-ramifiée si ni aucun idéal premier de $\OR_{\K}$ ni aucune injection réelle de $\K$ n'est ramifié dans $\Lc$.
\end{Def}

\begin{The}[\cite{Cox}(Th 5.18)]
	
	Soit $\K$ un corps de nombre. Il existe une unique extension $\HR/\K$ abélienne, non-ramifiée et maximale au sens de l'inclusion, c'est à dire telle que toute extension abélienne non-ramifiée de $\K$ est incluse dans $\HR$
\end{The}

Cette extension de $\K$ est liée au groupe de classe de $\OR_{\K}$ par le théorème de réciprocité d'Artin. L'énoncé exacte de ce théorème demande la construction du symbole d'Artin, qui généralise le symbole de Legendre pour les idéaux premiers (voir \cite{Cox}[Lem 5.19, Cor 5.21]). On énonce ici uniquement l'isomorphisme qui découle de ce théorème, et qui nous sera utile par la suite. 

\begin{The}[Théorème de réciprocité d'Artin pour le corps de classes\cite{Cox}(Th 5.23)]\label{CHilbert}
	
	Soit $\K$ un corps de nombre, $\HR$ son extension abélienne non-ramifiée maximale. Alors le groupe de Galois de l'extension $\HR/\K$ est isomorphe au groupe de classe $Cl(\OR_{\K})$
	\begin{equation*}
		Gal(\HR/\K)\simeq Cl(\OR_{\K})
	\end{equation*}
\end{The} 

\begin{Def}
	Soit $\K$ un corps de nombres. Son extension abélienne non-ramifiée maximale $\HR$ est appelé corps de classe de Hilbert de $\K$.
\end{Def}



\subsection{Algorithmes pour les réseaux structurés quadratiques}

Un réseau structuré est un idéal fractionnaire d'un ordre d'un corps de nombres. Le but de cette section est d'exposé les méthodes et les algorithmes utilisés pour manipuler les réseaux structurés quadratiques, pour lesquels le problème du vecteur le plus court est résolue en temps polynomial. 

\subsubsection{Forme standard d'un idéal et formes quadratiques binaires}

On énonce une correspondance entre d'une part les idéaux d'un ordre d'un corps quadratique imaginaire $\OR_{\Delta}$, et d'autre part certaines formes quadratiques définies positives de discriminant $\Delta$. Cela sera en particulier utile pour mener des calculs dans les groupes de classes d'idéaux. 

\begin{Def}
	Soit $f = ax^2 + bxy + cy^2$ une forme quadratique. $f$ est dite binaire si elle est définie positive et à coefficients dans $\Z$. Son discriminant est $\Delta = b^2 - 4ac$. On notera $f = f(a,b,c)$ pour désigner une forme quadratique binaire, ou $f = f(a,b)$ si le discriminant $\Delta$ est connu. $f$ est dite primitive si ses coefficients $a$, $b$ et $c$ sont premiers entre eux.
\end{Def}

\begin{Prop}[\cite{Coh00}(Th 7.7)]
	
	Soit $f = f(a,b)$ une forme quadratique binaire de discriminant $\Delta$. L'application
	\begin{equation*}
		I : f(a,b)\mapsto\left[ a, \frac{-b + \sqrt{\Delta}}{2}\right] 
	\end{equation*}
	associe à $f$ un idéal de $\OR_{\Delta}$.
\end{Prop}

Pour définir la réciproque de l'application $I$, il faut pouvoir retrouver les coefficients $a$ et $b$ à partir d'une écriture adéquate de l'idéal $\LR$.

\begin{Prop}[\cite{Buchmann95}(2.5)] \label{formestandard}
	
	Tout idéal entier $\LR\subset\OR_{\Delta}$ s'écrit 
	\begin{equation*}
		\LR = m\left[ a , \frac{-b + \sqrt{\Delta}}{2}\right],
	\end{equation*}
	avec $m,a \in\N$, $b\in\Z$, $b^{2} = \Delta [4a]$, $-a<b\leq a$ et $c = (b^2 -\Delta)/4a$ avec $a, b, c$ premiers entre eux.
\end{Prop}

\begin{Rem}
	La proposition citée \cite{Buchmann95}(2.5) est énoncée dans le cas des corps quadratique réel, de discriminant positif. Mais à part une condition différente sur le coefficient $b$, la preuve s'adapte sans difficulté au cas imaginaire.
\end{Rem}

\begin{Def}
	Un idéal $\LR$ est donné sous forme standard s'il est donné par l'unique base définie par \ref{formestandard}. Lorsque $\Delta$ est connue, on notera $\LR = id(m,a,b)$ une écriture standard de $\LR$, selon les notations de \ref{formestandard}. On dira que $\LR$ est primitif si $\LR = id(1,a,b)$.
\end{Def}

\begin{Coro}
	Soit $\LR = id(m,a,b)$ un idéal entier de $\OR_{\Delta}$. Soit $c = \frac{\Delta - b^2}{4a}$. L'application
	\begin{equation*}
		F : id(m,a,b)\mapsto f(a,b) = ax^2 + bxy + cy^2
	\end{equation*}
	associe à $\LR$ une forme quadratique binaire de même discriminant. 
\end{Coro}

\begin{Prop}[\cite{Cox}(Th 7.7)]
	Les applications $I$ et $F$ induisent une bijection entre les idéaux entiers primitifs et les formes quadratiques binaires primitives.
\end{Prop}

On peut retrouver facilement la norme et l'inverse d'un idéal entier donné sous forme standard.

\begin{Prop}[\cite{Coh00}(5.2.5)]
	
	Soit $\LR = id(m,a,b)\subset\OR_{\Delta}$ un idéal entier sous forme standard. Soit $\LR' = id\left( \frac{m}{N(\LR)},a,-b\right) $ (qui est un idéal fractionnaire de $\OR_{\Delta}$). Alors
	\begin{enumerate}
		\item $N(\LR) = m^2a$
		\item $\LR\LR' = \left[ \gcd(a,b,c),\;\frac{-b + \sqrt{\Delta}}{2}\right]  $
		\item $\LR$ est propre si et seulement s'il est primitif, et dans ce cas $\LR\LR' = \OR_{\Delta}.$
	\end{enumerate}
\end{Prop}

\begin{Def}
	Deux formes quadratiques binaires sont équivalentes si leurs images par $I$ le sont. On note $Cl(\Delta)$ les classes d'équivalences de formes quadratiques binaires de discriminant $\Delta$.
\end{Def}

L'application $F$ induit une structure de groupe sur les formes quadratiques binaires. Ce morphisme passe au quotient et induit un isomorphisme de groupe entre $Cl(\OR_{\Delta})$ et $Cl(\Delta)$. Cependant la notion d'équivalence entre formes quadratiques binaires peut être définie directement, ainsi qu'une loi de groupe sur les formes et sur leurs classes, via le produit le Dirichlet \cite{Cox}[3.8].

\begin{Def}
	Soit $f = (a,b)$, $f' = (a', b')$ deux formes quadratiques binaires de même discriminant $\Delta$. On défini le produit de Dirichlet $f*f'$ de la façon suivante : Soit $B$ défini modulo $2aa'$ par les relations suivantes
	\begin{enumerate}
		\item $B = b [2a]$
		\item $B = b' [2a']$
		\item $B^2 = \Delta [4aa']$
	\end{enumerate}
	Alors $f*f' = (aa', B)$ est une forme quadratique binaire de discriminant $\Delta$.
\end{Def}

\begin{Prop}[\cite{Cox}(Th 7.7)]
	
	Avec les notation précédentes, $I(f*g) = I(f)I(g)$ et $F(\LR\LR') = F(\LR)*F(\LR')$. Les applications $I$ et $F$ induisent des isomorphismes réciproques entre $Cl(\OR_{\Delta})$ et $Cl(\Delta)$.
\end{Prop}

\begin{Rem}
	On peut donc calculer la forme standard d'un produit d'idéaux eux mêmes sous formes standards, en calculant un produit de Dirichlet.
\end{Rem}

\subsubsection{Réduction de formes quadratiques binaires et applications}


La proposition suivante donne plusieurs moyens de trouver des formes équivalentes à une forme quadratique binaire donnée. On peut alors chercher une forme équivalente aux coefficients les plus petits possibles. 

\begin{Prop}[\cite{Cox}(Thm 2.8)]\label{actionMatriceForme}
	Les changements de variables en $(x,y)$ suivant appliqués à une forme quadratique binaire $f(a,b,c)$ permettent d'obtenir $f' = f(a',b',c')$ équivalente à $f$. 
	\begin{enumerate}
		\item $(x,y)\mapsto (x - qy, y),\;\forall\;q\in\Z$, alors $f' = f(a,b - 2aq, c - bq + aq^2)$  
		\item $(x,y)\mapsto(-y,-x)$, alors $f' = f(c, -b, a) $
	\end{enumerate}
	Ces changements de variables correspondent à l'action de certaines matrices de $SL_{2}(\Z)$ sur les formes quadratiques binaires (voir \cite{Cox}[Eq 2.2]). Si $f$ est primitive alors $f'$ l'est aussi. 
\end{Prop}

\begin{Def}
	Soit $f = f(a,b,c)$ une forme quadratique binaire de discriminant $\Delta < 0$. $f$ est une dite réduite si $|b| \leq a \leq c$ et si de plus lorsque $a = c$, $b \geq 0$. 
\end{Def}

\begin{Prop}[\cite{Cox}(2.8)]\label{uniciteReduite}
	Soit $\OR_{\Delta}$ un ordre d'un corps quadratique imaginaire. Toute classe de $Cl(\Delta)$ contient une unique forme quadratique réduite, qui est primitive. 
\end{Prop}

La preuve de la proposition \ref{uniciteReduite} fournit un algorithme donnant l'élément réduit de la classe d'une forme quadratique binaire donnée. Les différentes étapes de l'algorithme utilise les changements de variables cités dans la proposition \ref{actionMatriceForme}. 

\begin{algorithm}[h]
	\caption{Réduction de formes quadratiques binaires}
	\label{ALGO reduction}
	\begin{algorithmic}[1]
		\Require Une forme quadratique binaire $f(a,b,c)$
		\Ensure Une forme quadratique réduite équivalente.
		\State $test \gets False$
		\While{test = False}
		\If{ $|b| > a$}
		\State Calculer $q$, $r$ tels que $b = 2aq + r$, $-a<r\leq a$
		\State $c \gets c - bq + aq^2$
		\State $b \gets r$
		\EndIf
		\State $test \gets True$
		\If{$a > c$}
		\State Échanger $a$ et $c$
		\State $b \gets -b$
		\State $test \gets False$
		\EndIf
		\If{$b < 0$}
		\If{$a = -b$ OU $a = c$}
		\State $b \gets -b$
		\EndIf
		\EndIf
		\EndWhile
		\State \Return $f(a,b,c)$
	\end{algorithmic}
\end{algorithm}


L'algorithme \ref{ALGO reduction} est polynomial en la taille des coefficients de la forme quadratique binaire donnée. Plus précisément on a le résultat suivant. 

\begin{Prop}[\cite{Coh00}(5.4.3)]\label{nbEtapesReduction}
	
	Soit $(a,b,c)$ une forme quadratique binaire primitive de discriminant $\Delta$. L'algorithme \ref{ALGO reduction} renvoie une forme de réduite de $(a,b,c)$ en utilisant au plus $2 + \log\left( \frac{a}{\sqrt{|\Delta|}} \right)$ divisions euclidiennes.
\end{Prop}

\begin{Coro}\label{COMPLEX reduction}
	Soit $f = f(a,b,c) = F(\LR)$ une forme quadratique binaire associé à un idéal $\LR$ d'un ordre $\OR_{\Delta}$. La complexité de l'algorithme \ref{ALGO reduction} appliqué à $f(a,b,c)$ est en 
	\begin{equation*}
		O\left( \log(a)\log\left( \frac{a}{\sqrt{|\Delta|}} \right)\right).
	\end{equation*}
	
\end{Coro}

\begin{proof}
	La complexité de chaque passage dans la boucle de la ligne $2$ est dominée par l'étape de la ligne $4$, qui correspond à une division euclidienne de la variable $b$ par la variable $2a$. Puisque la forme $f$ correspond à un idéal sous forme standard, initialement $|b| \leq a$, donc toutes les divisions euclidienne seront de complexité en $O(\log(a))$. On conclut avec la proposition \ref{nbEtapesReduction}.
\end{proof}


\begin{Rem}
	On peut vérifier avec l'algorithme \ref{ALGO reduction} que deux idéaux entiers sont dans la même classe : il suffit de réduire leurs formes quadratiques binaires associées et vérifier que l'on trouve le même résultat. 
\end{Rem}

On énonce plusieurs propriétés des coefficients d'une forme quadratique réduite.

\begin{Prop}[\cite{Cox}(2.12)]\label{tailleCoefReduit}
	Une forme quadratique réduite $f = f(a,b)$ de discriminant $\Delta$ vérifie 
	\begin{equation*}
		|b|\leq a\leq\sqrt{\frac{|\Delta|}{3}}.
	\end{equation*}
	On peut donc représenter chaque classe $Cl(\OR_{\Delta})$ en $O(\log(|\Delta|))$ bits. 
\end{Prop}

\begin{Def}
	Un entier $m\in\N$ est dit représenté par une forme quadratique binaire $f$ s'il existe $s$, $t$ deux entiers tels que $m = f(s,t)$.
\end{Def}

\begin{Prop}[\cite{Cox}(Th 7.7)]
	
	Un entier $m\in\N$ est représenté par une forme quadratique binaire primitive $f$ si et seulement s'il existe un idéal entier $\LR\sim I(f)$ tel que $N(\LR) = m$.
\end{Prop}

\begin{Prop}[\cite{Cox}(Thm 2.8, 2.9)]\label{représentationEntierMinimaux}
	
	Soit $f = f(a,b,c)$ une forme quadratique binaire primitive. Alors tout les entiers représentés par $f$ sont positifs. Si $f$ est réduite, alors $a$ et $c$ sont les deux plus petits entiers représentés par $f$, avec $f(1,0) = a$ et $f(0,1) = c$. En particulier $a$ et $c$ sont premier entre eux.  
\end{Prop}

\begin{Coro}
	Si $f = f(a,b,c)$ est une forme quadratique réduite, et si $f' = f(c,-b,a)$, alors $I(f)$ et $I(f')$ sont les deux idéaux entier de plus petites de normes de leur classe d'équivalence. 
\end{Coro}


Il sera utile par la suite de trouver des générateurs d'idéaux principaux à partir de leur forme standard. Soit $\LR = id(m,a,b)$ un idéal entier supposé principal, et soit $\beta$ un générateur. D'après \ref{NormesId}, $N(\LR) = a = N_{\Q}^{\K}(\beta)$, donc étant donné la forme standard, on connaît la norme d'un générateur. On se ramène au cas où $\LR = id(1,a,b)$ est primitif, quitte à multiplier par $m$ le résultat final. 

\begin{Prop}
	Soit $\LR = m[a , \frac{-b + \sqrt{\Delta}}{2}]$ un idéal de $\OR_{\Delta}$ sous forme standard. Soit $f = F(\LR) = f(a,b)$ sa forme quadratique binaire primitive associée. Alors 
	\begin{equation*}
		f(x,y) = N_{\Q}^{\K}\left( m\left( ax - y\frac{-b + \sqrt{\Delta}}{2}\right) \right) / N(\LR)
	\end{equation*}
\end{Prop}

\begin{proof}
	Cette égalité découle de la définition de la norme d'extension $N_{\Q}^{\K}$.
\end{proof}

\begin{Coro}
	Soit $\LR = (1,a,b)\subset\OR_{\Delta}$ un idéal primitif sous forme standard. $\LR$ est principal si et seulement si $1$ est représenté par $f$. De plus dans ce cas,
	\begin{equation*}
		\LR = \left( ax - y\frac{-b + \sqrt{\Delta}}{2}\right) \OR_{\Delta}.
	\end{equation*}
\end{Coro}

\begin{proof}
	$\LR$ est un idéal principal si et seulement s'il existe un élément de $\LR$ de norme égale à $N(\LR)$. Puisque $\LR = [a , \frac{-b + \sqrt{\Delta}}{2}]$, tout élément $\beta$ de $\LR$ est de la forme
	\begin{equation*}
		\beta = \left( ax - y\frac{-b + \sqrt{\Delta}}{2}\right),\; x,y\in\Z.
	\end{equation*}
	Alors $N_{\Q}^{\K}(\beta) = N(\LR) \iff f(x,y) = 1$ d'après ce qui précède.
\end{proof}

On cherche une représentation de $1$ par $f$. D'après la proposition \ref{représentationEntierMinimaux}, lorsque $f$ est une forme quadratique binaire qui représente $1$, sa forme réduite $r$ est de la forme $x^2 \pm xy + cy^2$, et $r(1,0) = 1$. En appliquant en sens inverse les changements de variables utilisés lors de la réduction, on peut calculer une représentation de $1$ par $f$. On en déduit l'algorithme \ref{ALGO idealprincipal} permettant de trouver un générateur d'un idéal principal. Ceci est l'algorithme utilisé par défaut dans Sage pour la recherche de générateur.  

\begin{algorithm}[h]
	\caption{Générateur d'un idéal principal donné sous forme standard}
	\label{ALGO idealprincipal}
	\begin{algorithmic}[1]
		\Require $(m,a,b)$ un idéal entier d'un ordre $\OR_{\Delta}$ supposé principal.
		\Ensure Un élément $\beta$ tel que $(m,a,b) = \beta\OR_{\Delta}$. 
		\State $f\gets ax^2 + bxy + cy^2$
		\State $r\gets$ résultat de l'algorithme \ref{ALGO reduction} appliqué à $f$
		\State $L\gets$ liste des changements de variables utilisés durant la réduction de $f$ (voir la proposition \ref{actionMatriceForme}).
		\State $(x,y)\gets(1,0)$
		\State $(x,y)\gets(x',y')$ obtenus en appliquant tout les changements de variables retenus dans $L$ dans le sens inverse d'apparition.
		\State $\beta\gets m\left( ax - y\frac{-b + \sqrt{\Delta}}{2}\right)$
		\State \Return $\beta$
	\end{algorithmic}
\end{algorithm}


\begin{Prop}\label{COMPLEX idealprincipal}
	Soit $(m,a,b) = \beta\OR_{\Delta}$ un idéal entier principal de $\OR_{\Delta}$ donné sous forme standard (en particulier $ma$ est la norme de $\beta$). L'algorithme \ref{ALGO idealprincipal} renvoie $\beta$ avec une complexité en
	\begin{equation*}
		O\left( \log(a)\log\left( \frac{a}{\sqrt{|\Delta|}} \right)\right).
	\end{equation*}
\end{Prop}


\begin{Rem}
	Certaines sources \cite{BrokerCharlesLauter}, \cite{JaoSoukharev} utilisent un algorithme dû à Cornacchia pour résoudre le problème de l'idéal principal. Ils énoncent une complexité en $O(\log(N(\beta)))$. Cependant aucune preuve n'est donné, ni aucune description exacte de l'algorithme utilisé. Après analyse du problème il semble que cette approche ne fonctionne pas dans tout les cas. Dans toute la suite une utilisera l'algorithme \ref{ALGO idealprincipal}, utilisé par défaut dans Sage. 
\end{Rem}

L'algorithme \ref{ALGO idealprincipal} résout en particulier le problème du vecteur le plus court dans un réseau structuré quadratique principal. On généralise l'approche pour trouver une base de vecteurs court à partir de la proposition \ref{représentationEntierMinimaux}.

\begin{Prop}[\cite{Clapoti}(Prop 2.1)]\label{correspondanceIdeqElement}
	Soit $\OR_\Delta$ un ordre quadratique imaginaire, et $\LR$ un idéal entier. Soit $\alpha\in\LR$. Alors $\frac{\bar\alpha}{N(\LR)}\LR$ est un idéal entier de norme $N_{\Q}^{\K}(\alpha)$. De plus, pour tout idéal entier $\LR'\neq\LR$ de $\OR_{\Delta}$,
	\begin{equation*}
		\LR'\sim\LR\iff\exists\alpha\in\LR,\;\bar{\LR}\LR' = \bar{\alpha}\OR_{\Delta}\iff\exists\alpha\in\LR,\;\LR' = \frac{\bar\alpha}{N(\LR)}\LR.
	\end{equation*}
\end{Prop}

\begin{Rem}
	La proposition précédente donne une correspondance entre les élément $\alpha$ d'un idéal entier de $\OR_{\Delta}$ et les idéaux entiers $\LR'$ de la même classe. Connaissant $\LR$ et $\LR'$ sous forme standard, on utilise l'algorithme \ref{ALGO idealprincipal} pour trouver $\bar{\alpha}$ générateur de $\bar{\LR}\LR'$. Connaissant $\LR$ sous forme standard et $\alpha\in\LR$, on calcule la forme standard de $\frac{\bar\alpha}{N(\LR)}\OR_{\Delta}$ puis celle de $\LR'$.
\end{Rem}

On en déduit l'algorithme \ref{ALGO baseVecteurCourt}, prenant en entrée un idéal $\LR$ entier primitif d'un ordre $\OR_{\Delta}$, qui calcule les deux éléments de $\LR$ correspondant aux deux idéaux entiers de normes minimales obtenues avec la proposition \ref{représentationEntierMinimaux}. Ils forment une base de $\LR$ car leurs normes sont des entiers premiers entre eux. Si $\LR = id(m,a,b)$ n'est pas primitif, il suffit de traiter le cas primitif puis multiplier les éléments obtenus par $m$. 

\begin{algorithm}[h]
	\caption{Base de vecteurs courts d'un idéal (primitif) sous forme standard}
	\label{ALGO baseVecteurCourt}
	\begin{algorithmic}[1]
		\Require $\LR = id(1,a,b)$ un idéal entier primitif d'un ordre quadratique imaginaire $\OR_{\Delta}$.
		\Ensure Deux éléments $\alpha$, $\beta\in\LR$ de normes minimales tels que $\LR = [\alpha,\beta]$. 
		\State $f\gets F(\LR)$
		\State $r = f(a',b',c')\gets$ résultat de l'algorithme \ref{ALGO reduction} appliqué à $f$
		\State $\LR_1\gets I(f(a',b',c'))$
		\State $\LR_2\gets I(f(c',-b',a'))$
		\State $\alpha\gets$ conjugué du générateur de $\bar{\LR}\LR_1$ (Algorithme \ref{ALGO idealprincipal}).
		\State $\beta\gets$ conjugué du générateur de $\bar{\LR}\LR_2$ (Algorithme \ref{ALGO idealprincipal}).
		\State \Return $(\alpha,\beta)$
		
		
	\end{algorithmic}
\end{algorithm}

\begin{Prop}\label{COMPLEX baseVecteursCourts}
	Soit $\LR = id(m,a,b)$ un idéal entier de $\OR_{\Delta}$ donné sous forme standard. Quitte à se ramener au cas primitif, L'algorithme \ref{ALGO baseVecteurCourt} renvoie une base de normes minimales $\alpha, \beta$ avec une complexité en
	\begin{equation*}
		O\left( \log(a)\log\left( \frac{a}{\sqrt{|\Delta|}} \right)\right).
	\end{equation*}
\end{Prop}

\begin{Rem}
	Certaines sources \cite{Pegasis} utilise l'algorithme LLL pour trouver des bases réduites d'un réseaux structurés quadratiques. L'avantage ici face à LLL est de retourné à coup sûr une base de normes minimales, en restant en complexité polynomial.   
\end{Rem}

\subsection{Isogénies de courbes elliptiques}

On suppose connues les notions de théorie des courbes elliptiques présentées dans le cours du master. Dans la suite, on énonce des résultats généraux sur les courbes elliptiques découverts et étudiés pendant le stage.

\subsubsection{Noyaux d'isogénies}

On commence par énoncer les notions utilisées.

\begin{Def}
	On désigne par $E$ un ensemble de courbes elliptiques de même $j$-invariant. Un modèle de courbe elliptique est une courbe elliptique explicite donnée par une équation. Pour préciser le choix d'un modèle de courbe de $E$, on le donne par son équation. 
\end{Def}

\begin{Rem}
	On considère, sauf mention contraire, des modèles de courbes elliptiques donnés par des équation de Weierstrass réduites $E : y^2 = x^3 + ax + b$. On notera $\OR$ ou $\OR_{E}$ le point neutre. 
\end{Rem}

\begin{Def}
	Soit $E$ et $E'$ deux courbes elliptiques. On désigne par $\phi : E\rightarrow E'$ un ensemble d'isogénies égales à isomorphisme prêt au départ et à l'arrivée. Une isogénie $\phi$ est sous forme explicite lorsque l'on spécifie un modèle de départ et d'arrivé. $\phi$ est un endomorphisme lorsque $E$ est $E'$ sont isomorphes (on parlera d'endomorphisme même si $E\neq E'$). 
\end{Def}

\begin{Rem}
	Lorsque l'on voudra calculer des isogénies, on voudra plus précisément calculer des formes explicites d'isogénies.  
\end{Rem}

\begin{Ex}
	\begin{enumerate}
		\item Soit $E : y^2 = x^3 + ax + b$ un modèle de courbe elliptique. Si $P\in E$, on désigne par $\tau_{P}$ la translation par $P$. On rappelle que ce n'est pas une isogénie. 
		\item Soit $E$ une courbe elliptique. On note $[m]$ l'endomorphisme de $E$ consistant en la multiplication scalaire des points par $m$. On note $E[m]$ l'ensemble des points de $m$-torsions de $E$ (la notation $E[m]$ est donc aussi définie à isomorphisme prêt si l'on ne précise pas de modèle).
		\item Si $\phi$ est une isogénie, on note $\hat{\phi}$ son isogénie duale.
	\end{enumerate}
\end{Ex}

\begin{Def}
	Soit $E$, $E'$ deux courbes elliptiques définies sur un corps $\K$. Soit $\phi : E\rightarrow E'$ est une isogénie explicite. On définit son application contravariante (ou Pullback) par
	\begin{equation*}
		\phi^{*} : \Kb(E') \rightarrow \Kb(E), f\mapsto f\circ\phi
	\end{equation*} 
\end{Def}

Lorsque deux courbes elliptiques $E$ et $E'$ sont défini sur un corps $\K$, on sait que l'on peut associé à une isogénie $\phi : E\mapsto E'$ une extension $\Kb(E)/\phi^{*}\left( \Kb(E')\right)$. Par exemple les notions de degré ou de séparabilité de $\phi$ sont définies par rapport à cette extension. Les résultats présentés ici sont obtenus en appliquant la théorie de Galois à cette extension. On démontre que le groupe de Galois $Gal\left( \Kb(E)/\phi^{*}\left( \Kb(E')\right)\right)$ est isomorphes au noyaux de $\phi$.

\begin{Rem}
	Si une isogénie $\phi : E\rightarrow E'$ est défini sur un corps $\K$ et que l'on se donne une forme explicite définie sur $\K$, alors le noyau de $\phi$ n'est pas forcément défini sur $\K$.
\end{Rem}

\begin{The}[\cite{Sil09}(III 4.10)]
	
	Soit $E$, $E'$ deux courbes elliptiques défini sur un corps $\K$, et $\phi : E\rightarrow E'$ une isogénie non nulle. 
	\begin{enumerate}
		\item L'application 
		\begin{equation*}
			\ker(\phi) \rightarrow Aut\left( \left[ \Kb(E) : \phi^{*}\left( \Kb(E')\right) \right] \right) , P\mapsto \tau_{P}^{*}
		\end{equation*} 
		est un isomorphisme.
		\item Si $\phi$ est séparable, l'extension $\Kb(E)/\phi^{*}\left( \Kb(E')\right)$ est galoisienne, i.e $deg(\phi) = |Ker(\phi)|$
	\end{enumerate}
\end{The}

\begin{Coro}[\cite{Sil09}(III 4.12)]\label{unicitéIsogénieIso}
	Soit $E$ une courbe elliptique et $G$ un sous groupe fini de $E$. Il existe une unique courbe elliptique $E'$ et une isogénie séparable $\phi : E \rightarrow E'$ telles que $Ker(\phi) = G$.
\end{Coro}

\begin{Rem}
	On fera attention à l'unicité dans ce corollaire : il n'y a pas unicité de la forme explicite. En pratique le modèle de départ sera donné, et il faudra fixer un modèle d'arrivé, et expliciter un moyen de calculer l'image d'un point $P$ du modèle de départ. Un objectif de cette section est de définir une convention pour choisir les formes explicites. 
\end{Rem} 

\begin{Def}
	Soit $E$ une courbe elliptique et $G$ un sous groupe fini de $E$. L'unique courbe elliptique image d'une isogénie de noyau $G$ est noté $E/G$.
\end{Def}

Lorsque le noyau $G$ est donné comme sous groupe d'un modèle de courbe elliptique, on peut calculer une version explicite de l'isogénie de noyau $G$ avec une formule dû à Vélu donnant un modèle d'arrivé et l'image d'un point $P$:

\begin{Prop}[Formule de Vélu\cite{Velu71}]\label{FormuleVélu}
	
	Soit $E : y^2 = x^3 + ax + b$ un modèle de courbe elliptique défini sur un corps $\K$. Soit $G\subset E(\Kb)$ un sous-groupe fini. L'isogénie $\phi : E\rightarrow E/G$ de noyaux $G$ admet une forme explicite partant du modèle $E$, où l'image d'un point $P\in E(\Kb)$ est définie par :
	
	\begin{equation*}
		\phi(P) = \left( x(P) + \sum_{Q\in G\setminus{\OR}} x(P+Q) - x(Q)\;,\;y(P) + \sum_{Q\in G\setminus{\OR}} y(P+Q) - y(Q)\right) 
	\end{equation*} 
	
	Et le modèle d'arrivé est donné par l'équation $E' : y^2 = x^3 + a'x + b'$ avec
	\begin{equation*}
		a' = a - 5\sum_{Q\in G\setminus{\OR}}(3 (x(Q))^2 + a )
	\end{equation*}
	
	\begin{equation*}
		b' = b - 7\sum_{Q\in G\setminus{\OR}}(5 (x(Q))^3 + 3ax(Q) + b )
	\end{equation*}
\end{Prop}

Lorsqu'une courbe elliptique est définie sur un corps $\K$, on souhaitera calculer des formes explicites d'isogénies définies sur $\K$. On énonce un critère sur $G$ pour assurer l'existence de telles formes explicites.

\begin{Def}
	Soit $E : y^2 = x^3 + ax + b$ un modèle de courbe elliptique définie sur un corps $\K$. Soit $G$ un sous groupe fini de $E(\Kb)$, et $P\in E(\Kb)$. Le groupe $Gal(\Kb/\K)$ agit sur $P$ et $G$ de la façon suivante
	\begin{align*}
		\forall\sigma\in Gal(\Kb/\K),\;& P^{\sigma} = \left( \sigma(x(P)), \sigma(y(P))\right)\\
		& G^{\sigma} = \left\lbrace P^{\sigma},\; P\in G \right\rbrace 
	\end{align*}
\end{Def}

\begin{Prop}[\cite{Sil09}(II 4.12, 4.13.2)]
	
	Soit $E$ et $E'$ deux courbes elliptiques définies sur un corps $\K$. Une isogénie $\phi : E\rightarrow E'$ est définie sur $\K$ si et seulement si son noyau est stable sous l'action de $Gal(\Kb/\K)$, c'est à dire
	\begin{equation*}
		\forall\sigma\in Gal(\Kb/\K),\; \ker(\phi)^{\sigma} \subset \ker(\phi)
	\end{equation*}
\end{Prop}

On s'intéresse maintenant aux isogénies séparables de degré donné, c'est à dire dont on connaît le cardinal du noyau.

\begin{Def}
	On dira qu'une $\phi : E\rightarrow E'$ entre deux courbe elliptique est une $l$-isogénie si elle est de degré $l$. 
\end{Def}

\begin{Prop}
	Si $\phi : E\rightarrow E'$ est une $l$-isogénie entre les courbes elliptiques $E$ et $E'$, alors $\ker(\phi)\subset E\left[ l\right] $
\end{Prop}

\begin{proof}
	On rappelle que $\left[ l\right] = \hat{\phi}\circ\phi$. Ainsi si $P$ est un point d'un modèle de $E$, et si $P\in Ker(\phi)$, alors $lP = \hat{\phi}\circ\phi(P) = \hat{\phi}(\OR) = \OR $. Donc $P$ est un point de $l$-torsion de son modèle.
\end{proof}

On en déduit le nombre exacte de $l$-isogénie séparable sur une courbe elliptique $E$ :

\begin{Prop}\label{NbDeLIsogenies}
	Soit $E$ une courbe elliptique et $l\in\N$.
	\begin{enumerate}
		\item(rappel) $E[l] \simeq \Z/l\Z\times\Z/l\Z$
		\item Il existe exactement $l+1$ sous groupes d'ordre $l$ dans $\Z/l\Z\times\Z/l\Z$.
		\item Il existe exactement $l+1$ $l$-isogénie séparable sur $E$.
	\end{enumerate}
\end{Prop}


\begin{proof}
	Soit $P$, $Q$ une base de $E\left[ l\right]$ comme $\Z/l\Z$ module. $P$ et $Q$ sont ordre $l$ et sont libre sur $\Z/l\Z$, donc $\forall i\in\left[ 0 , l - 1\right] $, $P + iQ$ est d'ordre $l$. Ces points engendres des groupes disjoints (à l'exception du neutre $\OR$). On a donc trouvé $l + 1$ sous groupes d'ordre $l$ qui n'ont deux à deux que le neutre en commun. Ils regroupent au total de $1 + (l+1)*(l - 1) = l^2$ points distincts. Comme $E[l]$ est de cardinal $l^2$, on a trouvé exactement tout les sous groupes d'ordre $l$. Chacun d'eux défini exactement une isogénie séparable sur $E$.
\end{proof}





\subsubsection{Invariant différentiel et normalisation d'isogénies}



Pour fixer une convention sur le choix de formes explicites d'isogénie, on utilise la notion d'invariant différentiel. On commence par la définir sur une courbe algébrique de genre quelconque. Dans toute la suite, $C$ désigne une courbe algébrique projective supposée lisse, et $\Kb(C)$ l'ensemble de fonction rationnel sur $C$. 

\begin{Def}
	L'ensemble des formes différentielles de $C$, noté $\Omega_{C}$, est un $Kb(C)$-espace vectoriel engendré par des symboles $dx$ où $x$ parcours $\Kb(C)$.
	$dx$ est une dérivée formelle : on demande que trois relations soit vérifiée
	\begin{enumerate}
		\item $d(x+y) = dx + dy$ pour tout $x, y \in\Kb(C)$
		\item $d(xy) = xdy + ydx$ pour tout $x, y \in\Kb(C)$
		\item $da = 0$ pour tout $a\in\Kb$
	\end{enumerate}
\end{Def}

\begin{Prop}[\cite{Sil09}(II 4.2)]
	
	\begin{enumerate}
		\item $\Omega_{C}$ est de dimension 1 sur $\Kb(C)$
		\item Pour $x \in \Kb(C)$, $dx$ est une base de $\Omega_{C}$ si et seulement si l'extension $\Kb(C) / \Kb(x)$ est fini et séparable. 
	\end{enumerate}
\end{Prop}


\begin{Rem} On sait que l'extension $\Kb(C) / \Kb(x)$ est fini pour tout $x\in\Kb(C)$ non constant (d'après le théorème de normalisation de Noether, vu dans le cours de courbes algébriques). En caractéristique 0, la séparabilité de l'extension est garantie donc $dx$ est toujours une base de $\Omega_{C}$  
\end{Rem}

On peut étendre l'application contravariante d'un morphisme de courbe algébrique $\phi$ à aux formes différentielles. Cela donne notamment un critère de séparabilité qui nous sera utile par la suite.

\begin{Def}
	Soit $\phi : C \rightarrow C'$ un morphisme entre courbes algébriques $C$ et $C'$. On défini l'application contravariante sur $\Omega_{C}$ de la façon suivante :
	\begin{equation*}
		\phi^{*} : \Omega_{C'} \rightarrow \Omega_{C},\;\phi^{*}\left(\sum f_idx_i \right) = \sum \left( \phi^{*}(f_i) \right) d\left( \phi^{*}(x_i)\right) 
	\end{equation*}
\end{Def}

\begin{Prop}[\cite{Sil09}(II 4.2)]
	Soit $\phi : C \rightarrow C'$ un morphisme entre courbes algébriques $C$ et $C'$. $\phi$ est séparable si et seulement si $\phi^{*} : \Omega_{C'} \rightarrow \Omega_{C}$ est injective, ce qui équivaut à ce que $\phi^{*}$ soit non nulle. 
\end{Prop}

Sur une courbe algébrique lisse $C$, $f\in\Kb(C)$ vérifie $div(f) = 0$ si et seulement si $f$ est constante. Lorsque l'on généralise la notion de diviseur aux formes différentielles, certaines vont avoir un diviseur nul : ceux seront les invariants différentiels. 

\begin{Prop}[\cite{Sil09}(II 1.4)]
	Soit $C$ une courbe algébrique lisse. Soit $P\in C$ et $u\in \Kb(C)$ une uniformisante en $P$. Alors $du$ est une base de $\Omega_{C}$. 
\end{Prop}

\begin{Def}
	Soit $C$ une courbe algébrique lisse et $\omega\in\Omega_{C}$. Soit $P\in C$ et $u\in \Kb(C)$ une uniformisante en $P$. Il existe $g\in\Kb(C)$ tel que $\omega = gdu$. On note $\frac{\omega}{du} = g$
\end{Def}

\begin{Prop}[\cite{Sil09}(III 4.3)]
	Soit $C$ une courbe algébrique lisse et $\omega\in\Omega_{C}$. Soit $P\in C$ et $u\in \Kb(C)$ une uniformisante en $P$. L'ordre en $P$ de $\frac{\omega}{du} = g$ ne dépend pas du choix d'uniformisante $u$. On note donc $ord_{P}(\omega) = ord_{P}(\frac{\omega}{du})$. Alors $ord_{P}(\omega)$ est nul sauf pour un nombre fini de points $P$.
\end{Prop}


\begin{Def}
	Avec les notation précédentes, On défini $div(\omega) = \sum_{P\in C} ord_{P}(\omega)(P) \in Div(C)$. On dira que $\omega\in\Omega_{C}$ est holomorphe si $div(\omega)\geq 0$, et sans zéros si $div(\omega)\leq 0$. Un invariant différentiel $\omega$ est une forme différentielle holomorphe sans zéros, c'est à dire telle que $div(\omega) = 0$.
\end{Def}

\begin{Rem}
	On peut montrer que si $f\in\Kb(C)$ et $\omega\in\Omega_{C}$ alors $div(f\omega) = div(f) + div(\omega)$. Cela implique que tout les diviseurs de formes différentielles ont même classe dans $Pic(C)$, que l'on appelle la classe canonique. S'il existe un invariant différentiel, alors la classe canonique est la classe trivial. On peut en déduire avec le théorème de Riemann-Roch que s'il existe un invariant différentiel sur $C$, alors $C$ est de genre $1$. Réciproquement toutes les courbes elliptiques admettent un invariant différentiel.
		
\end{Rem}


\begin{Prop}[\cite{Sil09}(II 4.6)]
	Soit $E : y^2 = x^3 + ax + b$ un modèle courbe elliptique. Alors $\omega = \frac{dx}{2y}\in\Omega_{E}$ est un invariant différentiel de $E$.
\end{Prop}

\begin{Def}
	Soit $E : y^2 = x^3 + ax + b$ un modèle courbe elliptique. On note $\omega_{E}$ l'invariant différentiel défini par $\omega_{E} = \frac{dx}{2y}$. 
\end{Def}

Tout les invariants différentiels d'une courbe elliptique $E$ sont égaux à une constante multiplicative prêt. Étant donné un modèle $E$ de courbe elliptique d'invariant différentiel $\omega_{E}$, il est facile de trouver un modèle isomorphe $E'$ d'invariant différentiel $u\omega_{E}$ pour tout $u\in\Kb$

\begin{Prop}[\cite{Sil09}(III table 3.1)]\label{correctionInvariantDiff}
	
	Soit $E : y^2 = x^3 + ax + b$ un modèle courbe elliptique définie sur un corps $\K$, et $u\in\Kb$. Le modèle $E' : y^2 = x^3 + a'x + b'$ obtenue à partir de $E$ par le changement de variable $(x,y)\mapsto(u^{2}x,u^{3}y)$ est isomorphe à $E$ et vérifie $\omega_{E'} = u\omega_{E}$.
\end{Prop}


\begin{Def}	
	Soit $\phi : E\rightarrow E'$ une isogénie sous forme explicite, $\omega_{E}$ et $\omega_{E'}$ les invariants différentiels associés respectivement aux modèles de départ et d'arrivé. On dira que cette forme explicite de $\phi$ est normalisée si $\phi^{*}(w_{E'}) = w_E$	
\end{Def}

On énonce alors une version explicite du corollaire \ref{unicitéIsogénieIso}.

\begin{Prop}\label{unicitéIsogénieModèles}
	Soit $E : y^2 = x^3 + ax + b$ un modèle courbe elliptique et $G$ un sous-groupe fini de $E$. Soit $\phi : E\rightarrow E/G$ l'isogénie de noyau $G$. La courbe elliptique $E/G$ admet un unique modèle $E' : y^2 = x^3 + a'x + b'$ tel que la forme explicite de $\phi$ de départ $E$ et d'arrivé $E'$ soit normalisée. 
\end{Prop}

\begin{Rem}
	La formule de Vélu \ref{FormuleVélu} énoncée ne fournit pas une forme explicite normalisée a priori.
\end{Rem}




\subsubsection{Anneaux d'endomorphismes}

\begin{Def}
	Soit $E$ une courbe elliptique définie sur un corps $\K$. On note $End(E)$ l'anneau d'endomorphisme de $E$
\end{Def}

\begin{Rem}
	Si $E : y^2 = x^3 + ax + b$ et $E' : y^2 = x^3 + a'x + b'$ sont deux modèles isomorphes, une isogénie ayant une forme explicite $\phi : E\rightarrow E'$ est un endomorphisme, même si les modèles $E$ et $E'$ sont différents. Il sera souvent nécessaire pour normaliser d'écrire un endomorphisme sous une forme explicite où les modèles de départ et d'arrivé sont distincts. 
\end{Rem}

Il n'existe que $3$ structure possible pour un anneau d'endomorphisme de courbe elliptique.

\begin{The}[\cite{Sil09}(III 9.4)]
	
	Soit $E$ une courbe elliptique définie sur un corps $\K$. L’anneau $End(E)$ est isomorphe soit à $\Z$, soit à un ordre d'un corps quadratique imaginaire, soit à un ordre dans une algèbre de quaternions.
\end{The}

\begin{Rem} Soit $E$ une courbe elliptique définie sur un corps $\K$.
	\begin{itemize}
		\item Si $\K$ est de caractéristique nulle, $End(E)$ est un anneaux commutatif, donc soit isomorphe à $\Z$, soit à un ordre d'un corps quadratique imaginaire.
		\item Si $\K$ est fini de caractéristique $p>0$, $End(E)$ n'est jamais isomorphe à $\Z$.
	\end{itemize}
\end{Rem}

  
\begin{Def}
	Une courbe elliptique $E$ est dite à multiplication complexe si $End(E)$ n'est pas isomorphes à $\Z$. Si $E$ est définie sur un corps fini, elle est dite ordinaire si $End(E)\sim \OR_{\Delta}$ un ordre quadratique imaginaire de discriminant $\Delta$, et supersingulière sinon.  
\end{Def}

L'action de l'application contravariante d'une isogénie sur un invariant différentiel $\omega_{E}$ va s'avérer très simple, ce qui sera utile par la suite pour étudier les endomorphismes. 

\begin{The}[\cite{Sil09}(III, 5.2)]
	Soit $E$, $E'$ deux courbes elliptiques définie sur un corps $\K$, et $\phi$, $\psi : E \rightarrow E'$ deux isogénie. Soit $\omega$ un invariant différentiel d'un modèle de $E'$. Alors
	\begin{equation*}
		\left( \phi + \psi\right) ^{*}\left( \omega\right) = \phi^{*}\left( \omega\right) + \psi^{*}\left( \omega\right)
	\end{equation*} 
\end{The}


\begin{Coro}[\cite{Sil09}(III, 5.3)]
	Soit $E$ une courbes elliptiques sur $\K$, $\omega$ un invariant différentiel d'un modèle de $E$. Alors pour $m\in\Z^{*}$,
	$[m]^{*}(\omega) = m\omega$. 
\end{Coro}


\begin{Coro}[\cite{Sil09}(III, 5.6)]\label{LinéarisationEndomorphisme}
	
	Soit $E$ une courbe elliptique sur $\K$, $\omega$ un invariant différentiel d'un modèle de $E$. Pour tout endomorphisme $\phi\in End(E)$, soit $a_{\phi}\in\Kb(E)$ tel que $\phi^{*}(\omega) = a_{\phi}\omega$. On considère
	
	\begin{equation*}
		f : End(E)\rightarrow\Kb(E) , \phi\longmapsto a_{\phi}
	\end{equation*}
	Alors
	\begin{enumerate}
		\item $a_{\phi}\in\Kb$ et $f: End(E)\rightarrow\Kb$ est un morphisme d'anneaux. 
		\item $\ker(f)$ est exactement l'ensemble des endomorphisme inséparable de $E$
		\item Si $car(K) = 0$ alors $End(E)$ est commutatif
	\end{enumerate}
	
\end{Coro}



\subsubsection{Courbes elliptiques complexes et tores}

Dans cette section, toutes les courbes seront définie sur $\Kb = \C$. On énonce une équivalence de catégorie entre les courbes elliptiques définie sur $\C$ et les tores complexes, c'est à dire les quotients de $\C$ par un réseau. Il y a alors une correspondance entre les isogénies et les morphismes de tores, qui sont des homothéties. 

\begin{Prop}[\cite{Sil09}(VI 3.6)]
	Soit $\Lambda$ un réseau. Il existe une courbe elliptique, notée $E_\Lambda$, isomorphe (en tant que groupe de Lie complexe) au tore $\C/\Lambda$.
\end{Prop}

\begin{Rem}
	L'isomorphisme est rendu explicite par la fonction de Weierstrass associée au réseaux $\Lambda$. Pour plus de précision on se réfère à l'énoncé \cite{Sil09}(VI 3.6).
\end{Rem}

\begin{The}[Théorème d'uniformisation\cite{Sil09}(VI, 5.1, 5.2)]
	
	Soit $E$ une courbe elliptique définie sur $\C$. Il existe un réseau $\Lambda$ unique à homothétie prêt tel que $E = E_\Lambda$.
\end{The}


\begin{Def}
	Soit $\C/\Lambda_1$, $\C/\Lambda_2$ deux tores, et $\alpha\in\C$ tel que
	\begin{equation*}
		\alpha\in\left\lbrace \alpha\in\C : \alpha\Lambda_1 \subset \Lambda_2\right\rbrace  
	\end{equation*} 
	Un morphisme de tores est une application de la forme
	\begin{equation*}
		\phi_{\alpha} : \C/\Lambda_1 \rightarrow \C/\Lambda_2\; ,\; z \mapsto \alpha z
	\end{equation*}
\end{Def}

L'isomorphisme entre un réseau $\Lambda$ et sa courbe associée $E_\Lambda$ induit un isomorphisme entre anneaux d'endomorphismes.

\begin{The}[\cite{Sil09}(VI 4.1, 5.3)]
	
	Les catégories suivantes sont équivalentes
	\begin{enumerate}
		\item Les courbes elliptiques complexes, munies de leurs isogénies
		\item Les réseaux à homothéties prêt, munis des morphismes de tores
	\end{enumerate}
\end{The}

\begin{Coro}\label{EndoToresIsoEndoCourbes}
	Soit $\Lambda$ un réseau. Soit $R\in\C$ l'anneau des complexes représentant les endomorphisme du tore $\C/\Lambda$
	\begin{equation*}
		R = \left\lbrace \phi_{\alpha} :  \alpha\in\C ,\; \alpha\Lambda \subset \Lambda \right\rbrace 
	\end{equation*}
	
	Alors on a l'isomorphisme d'anneau $R\simeq End(E_{\Lambda})$
\end{Coro}


Lorsque $E$ est une courbe elliptique à multiplication complexe, $End(E)$ peut être inclus dans $\C$ de deux manières (à conjugaison prêt dans le corps quadratique imaginaire). On souhaite normaliser le choix de l'injection de $End(E)$ dans $\C$. Pour cela, on veut associer à tout $\alpha$ tel que $\alpha\Lambda\subset\Lambda$, un endomorphisme de $E$ notée $[\alpha]$. Le corollaire \ref{EndoToresIsoEndoCourbes} fourni une première convention. On va cependant définir une seconde convention de normalisation à l'aide de l'invariant différentiel et et du corollaire \ref{LinéarisationEndomorphisme}. Celle-ci se généralisera aux courbes elliptiques ordinaires sur corps finis, dans la section suivante. On montrera que les deux conventions de normalisation d'endomorphismes coïncident.

\begin{Def}
	Soit $E : y^2 = x^3 + ax + b$ un modèle d'une courbe elliptique $E$, et $\phi\in End(E)$. $\phi$ admet une unique forme explicite dont le modèle de départ et d'arrivé est $E : y^2 = x^3 + ax + b$. Il existe une constante $c\in\C$ telle que 
	\begin{equation*}
		(\phi)^{*}(\omega_E) = c\omega_E
	\end{equation*}
	La constante $c$ ne dépend pas du modèle de $E$ choisi, et est appelée facteur de l'endomorphisme $\phi$.
\end{Def}

On souhaite construire un morphisme d'anneau $\alpha\mapsto\left[ \alpha\right]\in End(E)$ de sorte que le facteur de $\left[ \alpha\right]$ soit $\alpha$. 

\begin{Prop}[\cite{Sil09}(III, 5.6)]
	Soit $E$ une courbe elliptique définie sur $\C$ à multiplication complexe, $\omega$ un invariant différentiel de $E$. Pour tout endomorphisme $\phi\in End(E)$, on note $a_{\phi}\in\Kb$ tel que $\phi^{*}(\omega) = a_{\phi}\omega$. Alors $a_\phi$ ne dépend pas du choix d'invariant différentiel. De plus le morphisme d'anneau
	\begin{equation*}
		f : End(E)\rightarrow\C , \phi\longmapsto a_{\phi}
	\end{equation*}
	est injectif.
\end{Prop}


Soit $\OR$ l'image de l'application $f$ de la proposition précédente. L'application $[.] : \OR \rightarrow End(E)$ recherchée est la réciproque de $f$. 

\begin{Prop}
	Soit $E$ une courbe elliptique sur $\C$ à multiplication complexe. Il existe un unique isomorphisme d'anneaux
	\begin{equation*}
		[.] : \OR\subset\C\rightarrow End(E)
	\end{equation*}
	tel que pour tout invariant différentiel $\omega$ de $E$,
	\begin{equation*}
		\forall\alpha\in\OR\;,\; [\alpha]^{*}(\omega) = \alpha\omega
	\end{equation*}
\end{Prop}


\begin{Def}
	Soit $E$ une courbe elliptique définie sur $\C$ à multiplication complexe. Soit $[.] : \OR\subset\C\rightarrow End(E)$ un isomorphisme d'anneaux. Le couple $(E,[.])$ est normalisé si pour tout invariant différentiel $\omega$ d'un modèle de $E$.
	\begin{equation*}
		\forall\alpha\in\OR\;,\; [\alpha]^{*}(\omega) = \alpha\omega
	\end{equation*}
\end{Def}

Avec cette convention de normalisation, l'image $\OR$ de $End(E)$ dans $\C$ correspond à l'anneau d'endomorphisme des tores équivalent à $E$.

\begin{Prop}[\cite{Sil94}(II 1.1)]
	Soit $(E,[.])$ une courbe elliptique définie sur $\C$ à multiplication complexe normalisé. Soit $[\alpha]$ un endomorphisme de facteur $\alpha$. Si $\Lambda$ est un réseau associé à $E$, alors $\alpha\Lambda\in\Lambda$ et l'endomorphisme $[\alpha]$ est équivalent à l'endomorphisme de tore $\phi_{\alpha}$.
\end{Prop}

La donnée d'un réseau associé à une courbe elliptique $E$ à multiplication complexe permet d'expliciter le corps quadratique imaginaire dont l'anneau d'endomorphisme est un ordre.

\begin{Prop}[\cite{Sil09}(VI 5.5)]
	
	Soit $\Lambda = \left[ z_{1}, z_{2}\right] $ un réseau , et $E_{\Lambda}$ sa courbe elliptique associée. Si $E_{\Lambda}$ est à multiplication complexe, alors $\K = \Q\left( \frac{z_1}{z_2}\right)$ est un corps quadratique imaginaire dont $End(E)$ est un ordre. 
\end{Prop}


\section{Introduction à la théorie de la multiplication complexe}



Dans les sections \ref{SECTION action} et \ref{SECTIONtorsion}, on introduit la théorie de la multiplication complexe. Les preuves seront rédigés. On se réfère à \cite{Sil94}(II), où les résultat sont énoncé pour des courbes elliptiques complexes, à multiplication complexe d'anneaux endomorphisme supposé être un ordre quadratique imaginaire maximal. On se place dans le cadre des ordres quadratiques imaginaires quelconques $\OR_{\Delta}$. Les preuves faites dans \cite{Sil94}(II) se généralise pour la plupart directement, en considérant toujours des idéaux propres. Lorsque $E$ est une courbe elliptique sur $\C$ à multiplication complexe, on supposera toujours l'isomorphisme $End(E)\simeq\OR_{\Delta}$ normalisé. Sauf mention contraire, $\K\subset\C$ est le corps quadratiques imaginaire dont $\OR_{\Delta}$ est un ordre.



\subsection{Action du groupe des classes d'idéaux}\label{SECTION action}


\begin{Def}
	On note $Ell_{\C}(\OR_{\Delta})$ l'ensemble des courbes elliptiques à multiplication complexes d'anneaux d'endomorphismes $\OR_{\Delta}$. 
\end{Def}

Si $\LR\in I(\OR_{\Delta})$, alors $\LR$ est un réseau de $\C$. On peut donc lui associer une courbe elliptique $E_{\LR}$.

\begin{Prop}[\cite{Sil94}(II, 1)]
	Soit $\OR_{\Delta}$ un ordre quadratique imaginaire, et $\LR\subset I(\OR_{\Delta})$. Alors $End(E_{\LR})\sim\OR_{\Delta}$, et $E_{\LR}\in Ell_{\C}(\OR_{\Delta})$.
\end{Prop}

\begin{proof}
	
	Par hypothèse, $\LR$ est un idéal fractionnaire propre, donc :
	\begin{align*}
		End(E_{\LR})&\simeq\left\lbrace\alpha\in\C : \alpha\LR\subset\LR\right\rbrace\\
		&= \left\lbrace\alpha\in\K : \alpha\LR\subset\LR\right\rbrace = \OR_{\Delta}
	\end{align*}
\end{proof}

Pour tout $c\in\K$ et $\LR\in I(\OR_{\Delta})$, $c\LR$ est à la fois un idéal équivalent à $\LR$ et un réseau homothétique à $\LR$. On s'intéresse donc aux classes d'idéaux. Soit $Cl(\OR_{\Delta})$ le groupe de classes de $\OR_{\Delta}$, et $\CL$ la classe de $\LR$. L'application
\begin{equation*}
	Cl(\OR_{\Delta})\rightarrow Ell_{\C}(\OR_{\Delta}),\; \CL\mapsto E_{\LR}
\end{equation*} 
est donc bien définie. On va montrer qu'il s'agit d'une bijection

\begin{Def}
	Soit $\Lambda$ un réseau et $\LR\in I(\OR_{\Delta})$. On défini le produit $\LR\Lambda$ par
	\begin{equation*}
		\LR\Lambda = \left\lbrace \alpha_1\lambda_1 + \ldots + \alpha_r\lambda_r : \alpha_i\in\LR , \lambda_i\in\Lambda , r\in\N\right\rbrace 
	\end{equation*}
\end{Def}

\begin{Prop}[\cite{Sil94}(II 1.2)]
	Soit $\Lambda$ un réseau et $\LR\in I(\OR_{\Delta})$.  Alors $\LR\Lambda$ est un réseau de $\C$. De plus, si $E_{\Lambda}\in Ell_{\C}(\OR_{\Delta})$ alors $E_{\LR\Lambda}\in Ell_{\C}(\OR_{\Delta})$
\end{Prop}

\begin{proof}
	Puisque $End(E_{\Lambda}) \sim \OR_{\Delta}$ est normalisé, d'après la proposition \ref{EndoToresIsoEndoCourbes}, $\OR_{\Delta}\Lambda = \Lambda$. De plus d'après \ref{denomid}, il existe $d\in\Z$ tel que $d\LR\subset\OR_{\Delta}$. Alors $\LR\Lambda\subset\frac{1}{d}\Lambda$, donc $\LR\Lambda$ est inclus dans un réseau de $\C$. Maintenant soit $a\in\Z\cap\LR$. Alors $a\OR_{\Delta}\subset\LR$ donc $a\Lambda\subset\LR\Lambda$. Ainsi $\LR\Lambda$ contient un réseau de $\C$. $\LR\Lambda$ est compris entre deux réseaux, donc est lui même un réseau de $\C$. Pour montrer que $E_{\LR\Lambda}\in Ell_{\C}(\OR_{\Delta})$, il suffit de montrer que $End(E_{\LR\Lambda}) = End(E_{\Lambda})$, ce qui est clair car $\LR$ est inversible. 
\end{proof}


\begin{Prop}[\cite{Sil94}(II,1.2)]\label{ActionClassesSTrans}
	Soit $\Lambda$ un réseau tel que $E_{\Lambda}\in Ell_{\C}(\OR_{\Delta})$, et $\LR$, $\LR'\in I(\OR_{\Delta})$. 
	\begin{enumerate}
		\item $E_{\LR\Lambda} = E_{\LR'\Lambda}$ si et seulement si $\CL = \CL'$
		\item Le groupe $C(\OR_{\Delta})$ agit sur $Ell_{\C}(\OR_{\Delta})$ par $\CL\cdot E_{\Lambda} = E_{\LR^{-1}\Lambda}$
		\item Cette action est simplement transitive. En particulier $\left| C(\OR_{\Delta})\right| = \left| Ell_{\C}(\OR_{\Delta})\right| $
	\end{enumerate}
\end{Prop}

\begin{proof}
	\begin{enumerate}
		\item $E_{\LR\Lambda} \sim E_{\LR'\Lambda}$ si et seulement s'il existe $c\in\C^{*}$ tel que $\LR\Lambda = c\LR'\Lambda$. $\LR$ et $\LR'$ étant inversible, cela équivaut à 
		\begin{equation*}
			\Lambda = c\LR^{-1}\LR'\Lambda = c^{-1}\LR'^{-1}\LR\Lambda
		\end{equation*}
		D'après la proposition \ref{EndoToresIsoEndoCourbes}, cela équivaut à ce que $c\LR^{-1}\LR'$ et $c^{-1}\LR'^{-1}\LR$ soit inclus dans $\OR_{\Delta}$. Étant inverse l'un de l'autre, le seul idéal entier de $\OR_{\Delta}$ dont l'inverse est aussi entier est $\OR_{\Delta}$ lui même. Donc cela équivaut à $c\LR^{-1}\LR' = \OR_{\Delta}$ et $\LR = c\LR'$. 
		
		\item L'action est bien définie sans difficulté, la présence de l'inverse dans la définition n'est pas nécessaire pour définir l'action, mais sera utile par la suite. 
		
		\item On se donne deux réseaux $\Lambda_1$ et $\Lambda_2$ dont les courbes elliptiques associées sont dans $Ell_{\C}(\OR_{\Delta})$, et on cherche $\CL$ tel que $\CL\cdot E_{\Lambda_1} = E_{\Lambda_2}$. Cela équivaut à trouver un idéal fractionnaire $\LR$ tel que $\Lambda_2$ et $\LR^{-1}\Lambda_1$ soient homothétiques. Mais puisque $E_{\Lambda_1}\in Ell_{\C}(\OR_{\Delta})$, $\OR_{\Delta}\Lambda_1 = \Lambda_1$ et 
		\begin{equation*}
			\left\lbrace \alpha\in\C : \alpha\Lambda_1\subset\Lambda_1\right\rbrace = \OR_{\Delta}
		\end{equation*}
		De plus 
		\begin{equation*}
			\forall \beta\in\C^{*}\;,\; \alpha\Lambda_1\subset\Lambda_1\iff\alpha\beta\Lambda_1\subset\beta\Lambda_1
		\end{equation*}
		Donc si l'on trouve $\beta$ tel que $\beta\Lambda_1\subset\K$, $\beta\Lambda_1$ sera un idéal fractionnaire propre de $\OR_{\Delta}$. Soit $z_1$, $z_2$ des générateurs de $\Lambda_1$. Si $\beta = z_1/z_2$ alors $\beta\Lambda_1\subset\K$. Et $\OR_{\Delta}\beta\Lambda_1 = \beta\Lambda_1$ donc $\beta\Lambda_1$ est un $\OR_{\Delta}$-module de type fini, donc c'est un idéal fractionnaire propre de $\OR_{\Delta}$. Soit $\LR_1 = \beta\Lambda_1$, et $\LR_2$ définie de la même manière à partir de $\Lambda_2$. Alors $\LR = \LR_2\LR_1^{-1}$ convient : $\LR^{-1}\Lambda_1$ est homothétique à $\Lambda_2$. on a donc démontré la transitivité, et la simple transitivité découle du point 1.
		
	\end{enumerate}
	
\end{proof}

L'application
\begin{equation*}
	Cl(\OR_{\Delta})\rightarrow Ell_{\C}(\OR_{\Delta}),\; \CL\mapsto E_{\LR}
\end{equation*} est donc une bijection. Notons que l'on peut normaliser les anneaux d'endomorphismes de toutes les courbes elliptiques de $Ell_{\C}(\OR_{\Delta})$ de façon cohérente.

\begin{Prop}[\cite{Sil94}(II, 1.1.1)]
	Soit $(E_1,[.]_1)$ et $(E_2,[.]_2)$ deux courbes elliptiques définies sur $\C$, à multiplication complexe, normalisées et de même anneaux d'endomorphisme $\OR_{\Delta}$. Soit $\phi : E_1 \rightarrow E_2$ une isogénie. Alors 
	\begin{equation*}
		\forall\alpha\in\OR_{\Delta},\;\phi\circ[\alpha]_1 = [\alpha]_2\circ\phi
	\end{equation*}
\end{Prop}

\begin{proof}
	Si $\omega$ est un invariant différentiel de $E_2$, alors $\phi^{*}(\omega)$ en est un pour $E_1$. L'endomorphsime $[\alpha]_{1}^{*}$ agit dessus par multiplication par $\alpha$. Ainsi
	\begin{align*}
		(\phi\circ[\alpha]_1)^{*}(\omega) &= [\alpha]_{1}^{*}(\phi^{*}(\omega))\\
		&= \alpha\phi^{*}(\omega)\\
		&= \phi^{*}(\alpha\omega)\\
		&= \phi^{*}([\alpha]_{2}^{*}(\omega))\\
		&= ([\alpha]_2\circ\phi)^{*}(\omega)\\
	\end{align*}
	Donc 
	\begin{equation*}
		(\phi\circ[\alpha]_1)^{*} = ([\alpha]_2\circ\phi)^{*}
	\end{equation*}
	D'après le corollaire \ref{LinéarisationEndomorphisme}, puisque sur $\C$ tout les endomorphismes sont séparables, 
	\begin{equation*}
		\phi\circ[\alpha]_1 = [\alpha]_2\circ\phi
	\end{equation*}
\end{proof}






\subsection{Groupes de torsions d'idéaux entiers}
\label{SECTIONtorsion}

Si $\phi : E \rightarrow E'$ est une isogénie entre deux courbes elliptiques $E$, $E'\in Ell_\C(\OR_{\Delta})$, alors $\phi$ correspond à l'action d'un unique élément $\CL\in Cl(\OR_{\Delta})$ sur $E$, puisque l'action est simplement transitive. On ne pourra pas déterminer $\phi$ à partir de $\CL$, car $\CL$ défini $\phi$ à endomorphisme prêt au départ et à l'arrivé. On souhaite définir à partir d'un idéal $\LR$ une isogénie $\phi : E\rightarrow \CL\cdot E$. Cela équivaut à définir le noyau de $\phi$ à partir de $\LR$. On n'y parviendra que dans le cas où $\LR$ est un idéal entier propre. 

\begin{Def}
	Soit $E\in Ell_\C(\OR_{\Delta})$, normalisée. Soit $\LR\in\OR_{\Delta}$ un idéal entier propre. Le sous-groupe de $\LR$-torsion d'un modèle de $E$ est 
	\begin{equation*}
		E\left[ \LR\right] = \left\lbrace P\in E : \left[ \alpha\right] (P) = \OR \; \forall\alpha\in\LR \right\rbrace 
	\end{equation*}
\end{Def}


\begin{Ex}
	\begin{enumerate}
		
		\item Si $\LR = m\OR_{\Delta}$ avec $m\in\Z$, alors $E\left[ \LR\right] = E\left[ m\right] $ est le groupe de $m$-torsion.
		
		\item Si $\alpha\in\OR_{\Delta}$, $E[\alpha\OR_{\Delta}] = \ker([\alpha])$ (l'égalité se démontre par double inclusions).
		
	\end{enumerate}
	
\end{Ex}

On montre que $E[\LR]$ est le noyau de l'isogénie $\phi : E\rightarrow \CL\cdot E$. La présence d'un inverse dans la définition de l'action $\CL\cdot E$ est nécessaire pour ce résultat.

\begin{Def}
	Soit $\LR\subset\OR_{\Delta}$ un idéal entier propre, et $\Lambda$ un réseau tel que $E_{\Lambda}\in Ell_{\C}(\OR_{\Delta})$. On associe à $\LR$ le morphisme de tores
	\begin{equation*}
		\phi_{\LR} : \C/\Lambda\rightarrow\C/\LR^{-1}\Lambda\; , \; z\mapsto z
	\end{equation*}
\end{Def}

\begin{Prop}[\cite{Sil94}(II,1.4)]\label{IsogénieIdealEntierIdentité}
	Soit $\LR\subset\OR_{\Delta}$ un idéal entier propre, et $\Lambda$ un réseau tel que $E_{\Lambda}\in Ell_{\C}(\OR_{\Delta})$. Alors $E_{\Lambda}\left[ \LR\right]$ est un sous-groupe fini de $E_{\Lambda}$, et la courbe $\CL\cdot E_{\Lambda}$ est l'unique courbe elliptique telle qu'il existe une isogénie $\phi : E_{\Lambda}\rightarrow \CL\cdot E_{\Lambda}$ de noyau $E_{\Lambda}\left[ \LR\right]$. Cette isogénie est celle associée par équivalence de catégorie à $\phi_{\LR}$, et sera encore notée $\phi_{\LR}$.
	
\end{Prop}


\begin{proof}
	On montre que $E_{\Lambda}\left[ \LR\right]$ est le noyau de $\phi_{\LR}$. D'abord, par normalisation des endomorphismes,
	\begin{align*}
		E_{\Lambda}\left[ \LR\right]&\sim \left\lbrace z\in\C/\Lambda : \alpha z = 0 \;,\; \forall\alpha\in\LR\right\rbrace \\
		&=\left\lbrace z\in\C : \alpha z\in\Lambda \;,\;\forall\alpha\in\LR\right\rbrace / \Lambda\\
		&=\left\lbrace z\in\C : z\LR\subset\Lambda\right\rbrace / \Lambda
	\end{align*}
	Or $\LR\Lambda = \Lambda$ car $\LR\subset\OR_{\Delta}$, donc $ \LR^{-1}\Lambda \subset \left\lbrace z\in\C : z\LR\subset\Lambda\right\rbrace $. Inversement si $z\in\C$ est tel que $z\LR\subset\Lambda$, alors $z\OR_{\Delta}\subset\LR^{-1}\Lambda$, donc en particulier $z\in\LR^{-1}\Lambda$. Donc $ \LR^{-1}\Lambda \subset \left\lbrace z\in\C : z\LR\subset\Lambda\right\rbrace $ et
	\begin{equation*}
		E_{\Lambda}\left[ \LR\right]\sim\LR^{-1}\Lambda/\Lambda = \ker(\phi_{\LR})
	\end{equation*}
	En particulier $E_{\Lambda}\left[ \LR\right]$ est un sous-groupe fini de $E$ en tant que noyau d'une isogénie non nulle. 
	
\end{proof}

Lorsque $\LR\subset\OR_{\Delta}$ est plus précisément un idéal premier avec le conducteurs, le degré de l'isogénie $\phi_{\LR}$ est la norme de $\LR$. 

\begin{Prop}[\cite{Sil94}(II, 1.4, 1.5)]
	Soit $\LR\subset\OR_{\Delta}$ un idéal premier avec le conducteur, et $\Lambda$ un réseau tel que $E_{\Lambda}\in Ell_{\C}(\OR_{\Delta})$. Alors :
	\begin{enumerate}
		\item $E\left[\LR\right]$ est un $\OR_{\Delta}/\LR$-module libre de rang $1$.
		\item $\phi_{\LR}$ est de degré $N\left( \LR\right)$ 
		\item Si $\alpha\in\OR_{\Delta}$, $[\alpha]$ est de degré $N_{\Q}^{\K}\left( \alpha\right) $
	\end{enumerate}
	
\end{Prop}

\begin{proof}
	
	\begin{enumerate}
		\item (Admis) La preuve de \cite{Sil94} utilise une version du théorème des restes chinois sur les idéaux fractionnaire, sur l'anneau des entiers $\OR_{\K}$. Ainsi elle ne se généralise aux ordres quelconques $\OR_{\Delta}$ qu'en se restreignant aux idéaux entier premier avec le conducteur.  
		
		\item Découle de 1. car $\left| E\left[\LR\right]\right| = \left| \OR_{\Delta}/\LR\right| = N\left( \LR\right)$.
		
		\item En particulier si $\LR = \alpha\OR_{\Delta}$ est principal, $[\alpha]$ est l'endomorphisme de noyau $E\left[\alpha\OR_{\Delta}\right]$. Donc 
		\begin{equation*}
			\deg \left( \left[\alpha \right] \right) = N\left( \alpha\OR_{\Delta}\right) = N_{\Q}^{\K}\left( \alpha\right) 
		\end{equation*}
	\end{enumerate}
\end{proof}


\subsection{Corps de définition de courbes elliptiques à multiplication complexe}\label{SectionCourbesSurHilbert}


Le but de cette section est de montrer que l'on peut réduire l'étude de $Ell_{\C}(\OR_{\Delta})$ à un sous-corps de $\Qb$. On se réfère toujours à \cite{Sil94}, mais les résultat seront admis. 

\begin{Def}
	Pour tout corps $\K$, on note $Ell_{\K}(\OR_{\Delta})$ l'ensemble des courbes elliptiques définies sur $\K$ d'anneaux d'endomorphisme $\OR_{\Delta}$.
\end{Def}


\begin{Def}
	Soit $E$ une courbe elliptique définie sur un corps $\K$. On note $j(E)\in\K$ son $j$-invariant. 
\end{Def}

\begin{Prop}[\cite{Sil09}(III, 1.4)]
	
	Soit $E$ une courbe elliptique définie sur $\C$. Alors $E$ est définie sur le corps $\Q\left( j\left( E\right) \right) $.
	
\end{Prop}


\begin{Def}
	Soit $E\in Ell_{\C}(\OR_{\Delta})$ et $\sigma$ un automorphisme de $\C$. Si $E : y^2 = x^3 + ax + b$ est un modèle de $E$, on définie le modèle $E^{\sigma} : y^2 = x^3 + \sigma(a)x + \sigma(b)$. Tout les modèles obtenues à partir de $\sigma$ et de modèles de $E$ sont isomorphes et définissent une courbe elliptique $E^{\sigma}$.
\end{Def}

\begin{Prop}[\cite{Sil94}(II 2.1)]
	
	Soit $E\in Ell_{\C}(\OR_{\Delta})$ et $sigma$ un automorphisme de $\C$. Alors
	\begin{enumerate}
		\item $End(E)\sim End(E^{\sigma})$
		\item $j(E)$ est algébrique sur $\Q$.
		\item $Ell_{\C}(\OR_{\Delta}) = Ell_{\Qb}(\OR_{\Delta})$
	\end{enumerate}
\end{Prop}


On n'aura pas besoin de se placer sur le corps $\Qb$ tout entier, mais seulement sur le corps de nombres engendré par les $j$-invariants de $Ell_{\Qb}(\OR_{\Delta})$. 

\begin{Prop}
	
	Soit $\OR_{\Delta}$ un ordre d'un corps quadratique imaginaire. Soit $\HR_{\Delta}\subset\Qb$ le sous-corps engendré par les $j$-invariants de $Ell_{\Qb}(\OR_{\Delta})$. Alors
	\begin{equation*}
		Ell_{\Qb}(\OR_{\Delta}) = Ell_{\HR_{\Delta}}(\OR_{\Delta})
	\end{equation*}
\end{Prop}


\begin{The}[\cite{Sil94}(II 4.3)]
	
	Pour tout corps quadratique imaginaire $\K$, et pour tout $E\in Ell_{\Qb}(\OR_{\K})$, $\Q\left( j\left( E\right) \right)$ est le corps de classe de Hilbert $\HR$ de $\K$. De plus, 
	\begin{equation*}
		Ell_{\Qb}(\OR_{\K}) = Ell_{\HR}(\OR_{\K})
	\end{equation*}
	
\end{The}

\begin{Rem}
	Ce théorème est en réalité plus précis. Comme $Cl(\OR_{\K})\simeq Gal(\HR/\K)$, on peut faire agir $Gal(\HR/\K)$ sur $Ell_{\Qb}(\OR_{\K})$. L'action est alors rendue explicite par le symbole d'Artin. C'est la raison principale pour laquelle la proposition \ref{ActionClassesSTrans} définie $\CL\cdot E_{\Lambda} = E_{\LR^{-1}\Lambda}$ avec un inverse. \cite{Sil94}[II 4.3.1]. De plus on montre que les $j$-invariants sont des entiers algébriques de $\OR_{\HR}$. Leurs polynômes minimaux sont dans $\Z[X]$, et trouver leurs racines dans $\C$ est une méthode très utilisé pour construire des courbes elliptiques à multiplication complexe (appelé "méthode CM").
\end{Rem}


\section{Représentations d'isogénies de courbes elliptiques ordinaires}

Une partie conséquente du stage consistait à comprendre comment la théorie de la multiplication complexe s'applique aux courbes elliptiques ordinaires, définies sur des corps finis. Plusieurs problèmes se posent : l'absence d'équivalence de catégories avec les tores complexes, et l'absence de bijection entre les endomorphismes et leurs facteurs. Pour palier à cela, dans la section \ref{SECTION ReducRelev} on énonce des résultats de réduction et de relèvement de courbes elliptiques. On démontre la proposition \ref{DefActionIdeauxSurCourbeOrdinaire}, qui justifie la définition d'une action de groupes de classes sur des courbes elliptiques ordinaires. Cette proposition, admise dans les articles cités, est à l'origine de tout les algorithmes de calculs d'isogénies étudiés. La section \ref{SECTION RepFrobenius} s'interresse à la description explicite d'endomorphismes dans le but de les calculer et les évaluer. Enfin la section \ref{SECTION volcanIsogenie} explicite comment utiliser la théorie de la multiplication complexe pour décrire la majorité des isogénies. 


\subsection{Réduction de courbes elliptiques et relèvement de Deuring}\label{SECTION ReducRelev}


Soit $E$ une courbe elliptique à multiplication complexe définie sur $\Qb$, et $p$ un nombre premier. On veut construire une courbe elliptique $E_p$ définie sur un corps fini $\F_{q}$ de caractéristique $p$, telle que $End(E)\simeq End(E_p)$.

\begin{Rem}
	La théorie générale de la réduction de courbes elliptiques utilise la notion de corps de nombres $p$-adiques. Cependant pour les courbes elliptiques à multiplication complexe, le fait de toujours être définie sur un corps de nombres permet de simplifier les énoncés. On n'énoncera pas la théorie générale.
\end{Rem}  

\begin{Def}
	On note $\Zb$ l'anneaux des entiers algébriques. Si $\PR$ est un idéal maximal de $\Zb$, on dira que $\PR$ est au dessus d'un nombre premier $p$ si $\PR\cap\Z = p\Z$. 
\end{Def}

\begin{Prop}
	Soit $E$ une courbe elliptique définie sur $\Qb$. Alors $E$ admet un modèle à coefficient dans $\Zb$. 
\end{Prop}

\begin{proof}
	Soit $E : y^2 = x^3 + ax + b$ un modèle de $E$ définie sur $\Qb$. Pour tout $u\in\C^{*}$, on peut se ramener à isomorphisme prêt au modèle 
	$E : y^2 = x^3 + u^4ax + u^6b$. Si $a$ et $b$ ne sont pas dans $\Zb$, puisque $\Qb$ est le corps de fraction de $\Zb$, en choisissant pour $u$ le ppcm des dénominateurs de $a$ et $b$, on se ramène à un modèle définie sur $\Zb$. 
\end{proof}

\begin{Def}
	Soit $\PR$ un idéal premier de $\Zb$, $E$ une courbe elliptique définie sur $\Zb$. On définit la réduction modulo $\PR$ d'un modèle de $E$ de la façon suivante : On se donne $E : y^2 = x^3 + ax + b$ un modèle de $E$ défini sur $\Zb$. Soit $k = \Zb / \PR$. Le modèle réduit $E_\PR$ est défini sur $k$ par l'équation $E_\PR : y^2 = x^3 + \tilde{a}x + \tilde{b}$. La courbe elliptique $E_\PR$ ne dépend pas du choix de modèle de $E$. On dira que la réduction est bonne si le discriminant du modèle réduit est non nul. On défini la réduction des points et des isogénies en considérant leurs coordonnées ou formes explicites modulo $\PR$. 
\end{Def}

\begin{Rem}
	Pour une définition plus générale et une démonstration de la validité de cette définition, on se réfère à \cite{Sil09}(VII 1). Si $E$ est définie sur $\Z$, et si l'idéal $\PR$ est au dessus d'un nombre premier $p$, la réduction modulo $\PR$ coïncide avec la réduction modulo $p$. Préciser l'idéal $\PR$ permet de définir la réduction des points et des isogénies qui ne sont pas défini sur $\Q$.  
\end{Rem}

\begin{Rem}\label{RemarqueReductionCorpsFinis}
	Soit $E\in Ell_{\HR_\Delta}(\OR_{\Delta})$ et $\PR\subset\Zb$ un idéal maximal au dessus de $p$. Alors $E_\PR$ est définie sur $\OR_{\HR_\Delta}/\left( \PR\cap\OR_{\HR_\Delta}\right) $. L'idéal $\PR\cap\OR_{\HR_\Delta}$ est au dessus de $p$ et premier dans $\OR_{\HR_\Delta}$, donc maximal et $\OR_{\HR_\Delta}/\left( \PR\cap\OR_{\HR_\Delta}\right)$ est un corps fini. $E_\PR$ est donc définie sur $\F_{q}$ où $q$ est déterminé par le degré d'inertie de $p$ dans $\HR_\Delta$. Notamment si $p$ est décomposé dans $\HR_\Delta$, $E_\PR$ est défini sur $\F_{p}$.
	
\end{Rem}

Il est possible que des points finis de $E$ se réduisent sur le point à l'infini de $E_\PR$. Cependant cela n'arrivera pas pour les points de $E$ d'ordres finis premier avec la caractéristique $p$.

\begin{Prop}\label{ReductionPointInjectivité}
	Soit $\PR$ un idéal premier de $\Zb$ au dessus d'un nombre premier $p$. Soit $E$ une courbe elliptique définie sur un corps de nombre $\HR$ ayant une bonne réduction modulo $\PR$. On note $k = \OR_{\HR}/\PR\cap\OR_{\HR}$. Soit $n$ premier avec la caractéristique $p$. L'application
	\begin{equation*}
		P\in E[n](\HR) \rightarrow \tilde{P}\in E_\PR[n](k)
	\end{equation*}
	est injective. En particulier $E[n]\simeq E_\PR[n]$
\end{Prop}

\begin{Rem}
	Pour la démonstration de cette proposition dans un cadre plus général, on se réfère à \cite{Sil09}(VII 3.1).
\end{Rem}

La réduction des isogénie est une application injective, mais pas surjective en général. 

\begin{Prop}\label{reductionIsogenieInjectivité}
	Soit $E$, $E'$ deux courbes elliptiques définies sur un corps de nombre $\HR$, qui admettent une bonne réduction modulo $\PR\subset\Zb$ idéal maximal. Soit $Hom(E, E')$ l'ensemble des isogénies entre $E$ et $E'$. L'application 
	\begin{equation*}
		Hom(E, E')\rightarrow Hom(E_\PR, E_\PR),\;\phi\mapsto\tilde{\phi}
	\end{equation*}
	est injective, et préserve le degré : $deg(\phi) = deg(\tilde{\phi})$
	
\end{Prop}

\begin{Rem}
	Pour une définition plus rigoureuse de la réduction d'isogénie et une démonstration de cette propriété, on se réfère à \cite{Sil94}(II, 4.4) et \cite{Lang1987}(9 \S2).
\end{Rem}

On cherche à savoir sous quelles conditions on obtient un isomorphisme $End(E)\simeq End(E_\PR)$. On cite pour cela un critère dû à Deuring :

\begin{The}[\cite{Lang1987}(Thm 13.12)]\label{CritèreReductionComplete}
	
	Soit $E$ une courbe elliptique à multiplication complexe, avec $End(E)$ ordre quadratique imaginaire de conducteur $f$, d'un corps $\K$. On note $f = p^r f_{0}$ avec $p$ premier ne divisant pas $f_{0}$. Soit $\PR\subset\Zb$ un idéal maximal au dessus de $p$.
	\begin{enumerate}
		\item Si $p$ est décomposé dans $\K$, alors $E_\PR$ est ordinaire, et $End(E_\PR)$ est un ordre de $\K$ de conducteur $f_{0}$.
		\item Si de plus $p$ de divise pas $f$, $End(E)\simeq End(E_\PR)$, et l'application 
		\begin{equation*}
			End(E)\rightarrow End(E_\PR),\;\phi\mapsto\tilde{\phi}
		\end{equation*} est un isomorphisme.
		\item Si $p$ est ramifié ou inerte dans $\K$, alors $E_\PR$ n'est pas ordinaire.
	\end{enumerate}
	
\end{The}


\begin{Def}
	Soit $E$ une courbe elliptique à multiplication complexe et $p$ un nombre premier. On dira que $E$ se réduit complètement l'idéal maximal $\PR\subset\Zb$ si elle admet une bonne réduction modulo $\PR$ et que de plus $End(E)\simeq End(E_\PR)$.
\end{Def}

Dans le cas d'une réduction complète, on peut étendre la normalisation de $End(E)$ à $End(E_\PR)$ en considérant les facteurs modulo $\PR$.

\begin{Prop}[\cite{Lang1987}(9 \S4)]\label{facteurEndomorphismeReduction}
	
	Soit $(E,[.])$ une courbe elliptique complexe normalisée d'anneau d'endomorphismes isomorphe à $\OR_{\Delta}$, et $\alpha\in\OR_{\Delta}$. Soit $\omega$ un invariant différentiel d'un modèle de $E$, $\tilde{\omega}$ l'invariant différentiel de $E_\PR$ en considérant $\omega$ modulo $\PR$. On suppose que $E$ se réduit complètement modulo $\PR\subset\Zb$. Alors 
	\begin{equation*}
		\left( \widetilde{\left[ a\right] }\right)^{*}\left( \tilde{\omega}\right) = \tilde{\alpha}\tilde{\omega}
	\end{equation*}
	En particulier $\widetilde{[\alpha]}$ est inséparable si et seulement si $\alpha\in\PR$
\end{Prop}

\begin{Rem}
	D'après la proposition précédente, la notation $\left[ \tilde{\alpha} \right]$ n'a donc pas de sens par il existe plusieurs endomorphismes de $E_\PR$ de facteur $\tilde{\alpha}$.
\end{Rem}



On se donne désormais une courbe elliptique $E$ ordinaire définie sur un corps fini $\F_{q}$. On souhaite voir $E$ comme la réduction d'une certaine courbe $E'$ définie sur $\Qb$, modulo un idéal maximal $\PR\subset\Zb$. Il faut d'abord que le critère du théorème \ref{CritèreReductionComplete} soit respecté. C'est en réalité toujours le cas.

\begin{The}[\cite{Lang1987}(13.5)]
	
	Soit $E$ une courbe elliptique ordinaire définie sur un corps fini $\F_{q}$ de caractéristique $p$. Ainsi $End(E)$ est un ordre quadratique imaginaire de conducteur $f$ d'un corps $\K$. Alors $p$ est premier avec $f$ et décomposé dans $\K$. 
	
\end{The}

Le théorème de relèvement de Deuring énonce que la surjectivité de la réduction. 

\begin{The}[Théorème de relèvement de Deuring\cite{Lang1987}(13.14)]\label{RelevementDeuring}
	
	Soit $E$ une courbe elliptique ordinaire définie sur un corps fini $\F_{q}$, et $\phi\in End(E)$ qui n'est pas une multiplication scalaire. Il existe une courbe elliptique $E'$ définie sur un corps de nombre $\HR$, munie d'un endomorphisme $\phi'$ et d'un idéal maximal $\PR\subset\Zb$ au dessus de $p$, tels que $E'_\PR = E$ et $\tilde{\phi'} = \phi$.
	
\end{The}

Il est possible de voir tout les éléments de $Ell_{\F_{q}}(\OR_{\Delta})$ comme des réductions complètes modulo un même idéal maximal $\PR\in\Zb$. En particulier la proposition \ref{DescriptionFrobReduction} implique que l'idéal $\PR\cap\OR_{\Delta}$ est déterminé par l'image du morphisme de Frobenius dans $\OR_{\Delta}$. 

\begin{Prop}[\cite{Lang1987}(Cor 13.5)]\label{DescriptionFrobReduction}
	
	Soit $E\in Ell_{\Qb}(\OR_\Delta)$, $End(E)$ étant un ordre du corps $\K$. On suppose que $E$ se réduit complètement modulo $\PR\subset\Zb$. Ainsi existe un corps finis $\F_{q}$ tel que $E_\PR$ y soit définie (Remarque \ref{RemarqueReductionCorpsFinis}), et tel que la caractéristique $p$ soit décomposé dans $\OR_{\Delta}$. Soit $\pi_{q}$ l'endomorphisme de Frobenius, et $\alpha\in\OR_{\Delta}$ tel que $\widetilde{[\alpha]} = \pi_{q}$. Alors $\alpha\in \PR\cap\OR_{\Delta}$. De plus l'un des deux idéaux au dessus de $p$ dans $\OR_{\Delta}$ est $\PR\cap\OR_{\Delta}$. Enfin si $q = p^d$, $\alpha\OR_{\Delta} = \left(\PR\cap\OR_{\Delta} \right) ^{d}$.
	
	
\end{Prop}

\begin{Prop}[\cite{BrokerCharlesLauter}(\S2)]
	
	Soit $\OR_{\Delta}$ un ordre d'un corps quadratique imaginaire $\K$ de conducteur $f$, et $\F_{q}$ un corps fini de caractéristique $p$. On suppose que $p$ de divise pas $f$ et est décomposé dans $\K$. Alors il existe un idéal $\PR\subset\Zb$ maximal tel que pour tout $E\in Ell_{\F_{q}}(\OR_{\Delta})$, il existe $E'\in Ell_{\Qb}(\OR_{\Delta})$, $\phi'\in End(E')$, $\phi\in End(E)$ tels que $E'_\PR = E$ et $\tilde{\phi'} = \phi$. Dans ce cas, la réduction modulo $\PR$ est une bijection entre $Ell_{\Qb}(\OR_{\Delta})$ et $Ell_{\F_{q}}(\OR_{\Delta})$.
\end{Prop}

L'action de $Cl(\OR_{\Delta})$ sur $Ell_{\Qb}(\OR_{\Delta})$ induit une action de $Cl(\OR_{\Delta})$ sur $Ell_{\F_{q}}(\OR_{\Delta})$.

\begin{Def}
	Soit $\PR\subset\Zb$ idéal maximal tel que la réduction modulo $\PR$ est une bijection entre $Ell_{\Qb}(\OR_{\Delta})$ et $Ell_{\F_{q}}(\OR_{\Delta})$. On défini une action de $Cl(\OR_{\Delta})$ sur $Ell_{\F_{q}}(\OR_{\Delta})$ de la façon suivante : Soit $E'\in Ell_{\Qb}(\OR_{\Delta})$, $E = E'_\PR$ et $\CL\in Cl(\OR_{\Delta})$. Alors
	\begin{equation*}
		\CL\cdot E = (\CL\cdot E')_\PR
	\end{equation*}
\end{Def}

\begin{Rem}
	Ainsi l'injectivité de la réduction des isogénies (proposition \ref{reductionIsogenieInjectivité}) permet d'associer à toute isogénie entre courbes de $Ell_{\Qb}(\OR_{\Delta})$ une isogénie entre leurs courbes réduites de $Ell_{\F_{q}}(\OR_{\Delta})$. On rappelle cependant que l'on n'a pas la surjectivité en général.
\end{Rem}

On montre alors que l'on peut généraliser la notion de groupe de torsion d'un idéal entier propre $\LR\subset\OR_{\Delta}$. Il s'agit de montrer que la réduction et l'action par un idéal commutent. 

\begin{Def}
	Soit $E'\in Ell_{\Qb}(\OR_{\Delta})$ normalisée, qui se réduit complètement modulo un idéal maximal $\PR\subset\Zb$. Soit $E = E'_\PR$. On associe à un idéal entier propre $\LR\subset\OR_{\Delta}$ un sous groupe de $E$, noté $E\left[ \LR\right]$, défini par
	\begin{equation*}
		E\left[ \LR\right] = \left\lbrace P\in E :\; \widetilde{\left[ \alpha\right] }(P) = 0 \;\forall\alpha\in\LR\right\rbrace .
	\end{equation*} 
	
\end{Def}

\begin{Prop}\label{DefActionIdeauxSurCourbeOrdinaire}
	Soit $E'\in Ell_{\Qb}(\OR_{\Delta})$ normalisée, qui se réduit complètement modulo  un idéal maximal $\PR\subset\Zb$. Soit $E = E'_\PR$, $\LR\subset\OR_{\Delta}$ un idéal entier propre, et l'isogénie
	\begin{equation*}
		\phi_{\LR} : E'\rightarrow E'/E'\left[ \LR\right] = \CL\cdot E'
	\end{equation*} 
	Alors le noyau de l'isogénie réduite $\widetilde{\phi_{\LR}}$ est $E\left[ \LR\right]$, et de plus $E/E\left[ \LR\right] = \CL\cdot E$.
\end{Prop}

\begin{proof}
	Les groupe de torsion sur $E'$ de tout idéal entier $\LR\subset\OR_{\Delta}$ sont finis. Donc il existe $m\in\N^*$ tel que $E'[\LR]\subset E'[m]$. D'après la proposition \ref{ReductionPointInjectivité}, la réduction de $E'[m]$ modulo $\PR$ est injective. Donc en particulier elle induit une injection de $E'[\LR]$ dans $E'_\PR[m]$. Soit $P\in E'[\LR]$ non nul. Alors 
	\begin{equation*}
		\tilde{P}\in \ker\left( \widetilde{\phi_{\LR}}\right)  \iff \widetilde{\phi_{\LR}(P)} = \tilde{\OR} \iff \phi_{\LR}(P) = \OR \iff P\in E'[\LR],
	\end{equation*}
	car $\phi(P)$ est d'ordre fini non nul donc sa réduction est non nulle. De plus
	\begin{equation*}
		\tilde{P}\in E[\LR] \iff \widetilde{\left[ \alpha\right] }\left( \tilde{P}\right)  = \widetilde{\left[ \alpha\right] (P)} =  0,\;\forall\alpha\in\LR \iff {\left[ \alpha\right] }(P) = 0, \;\forall\alpha\in\LR \iff P\in E'[\LR].
	\end{equation*}
	 Ainsi $\ker\left( \widetilde{\phi_{\LR}}\right)  = E[\LR]$. La courbe image de $\widetilde{\phi_{\LR}}$ est $\CL\cdot E$ par définition, donc 
	\begin{equation*}
		E/E\left[ \LR\right] = \CL\cdot E.
	\end{equation*}
\end{proof}










\subsection{Représentation complexe de l'endomorphisme de Frobenius}\label{SECTION RepFrobenius}

Soit $E$ une courbe elliptique définie sur un corps fini $\F_{q}$. Alors le morphisme de Frobenius $\pi_{q}$ est un endomorphisme de $E$ et $End(E)$ contient $\Z[\pi_{q}]$. Dans le cas où $E$ est ordinaire, ces deux ensembles sont des ordres d'un même corps quadratique imaginaire $\K$, et
\begin{equation*}
	\Z[\pi_{q}]\subset End(E)\simeq\OR_{\Delta}\subset\OR_{\K}.
\end{equation*} 

\begin{Def}
	Soit $E$ une courbe elliptique à ordinaire définie sur un corps fini $\F_{q}$. On note $f_{m}$ le conducteur de l'ordre $\Z[\pi_{q}]$. On notera $f$ le conducteur de $End(E)\simeq\OR_{\Delta}$, ou $f_{E}$ si besoin. On appellera $f$ le conducteur de $E$, et $f_{m}$ le conducteur maximal de $E$. 
\end{Def}

\begin{Prop}
	Soit $E$ une courbe elliptique définie sur un corps fini $\F_{q}$. Son conducteur $f$ divise son conducteur maximal $f_{m}$.
\end{Prop}

On va se servir du conducteur maximal pour représenter $\pi_{q}$ dans $\OR_{\Delta}\subset\C$, puis pour obtenir une base de l'ordre $End(E)$ à partir d'endomorphismes entiers et de $\pi_{q}$.

\begin{Prop}
	\label{lienEntierWkFq}
	Soit $E$ une courbe elliptique ordinaire définie sur $\F_{q}$ munit d'un isomorphisme $End(E)\simeq\OR_{\Delta}$. Soit $\alpha\in\OR_{\Delta}$ l'image du morphisme de Frobenius $\pi_{q}$.
	Alors $\alpha$ est une racine du polynôme caractéristique de $E$. Soit $\OR_{\K} = \left[ 1, w_{\K}\right] $ (voir proposition \ref{baseOrdreWk}). Alors il existe un entier $s$ tel que $\pm\alpha = f_{m}w_{\K} + s$
	
\end{Prop}

\begin{proof}
	Soit $X^{2} - tX + q$ le polynôme caractéristique de $E$. Il annule $\pi_{q}\in End(E)$. Donc $\alpha$ est racine de $X^{2} - tX + q$ de discriminant $\Delta_{m} = t^{2} - 4q = f_{m}^{2}d_{\K}$. Donc $\alpha = \frac{t\pm f_{m}\sqrt{d_{\K}}}{2}$, et $w_{\K} = \frac{d_{\K} + \sqrt{d_{\K}}}{2}$ par définition. On en déduis que  $\pm\alpha = f_{m}d_{\K} - \frac{f_{m}d_{\K} \pm t}{2}$. Or $t^{2} - f_{m}^{2}d_{\K} = 4q$ est pair, donc $t$ et $f_{m}d_{\K}$ ont même parité et $s = -\frac{f_{m}d_{\K} \pm t}{2}$ est un entier.
	
\end{proof}

\begin{Rem}
	Soit $q = p^d$. D'après le théorème de relèvement de Deuring \ref{RelevementDeuring}, il existe une courbe $\left( E',\left[ .\right] \right) $ définie sur $\C$ à multiplication complexe, munie d'un endomorphisme $\left[ \alpha\right] $ et d'un idéal maximal $\PR\subset\Zb$, tels que 
	\begin{equation*}
		E'_\PR = E,\;\;\widetilde{\left[ \alpha\right] } = \pi_{q},\;\;\alpha\OR_{\Delta} = \left(\PR\cap\OR_{\Delta} \right)^{d}
	\end{equation*}
	Alors $\alpha$ est l'une des deux racines complexes du polynôme caractéristique de $E$, mais sa valeur dépend de l'idéal de réduction $\PR$.
	
\end{Rem}

\begin{Def}
	Dans toutes la suite, on note indifféremment $\pi_{q}$ et son image $\alpha\in\OR_{\Delta}$. On suppose l'idéal de réduction $\PR$ choisit de sorte que $\pi_{q} = \frac{t + f_{m}\sqrt{d_{\K}}}{2}$. On note $s$ l'entier donné par la proposition \ref{lienEntierWkFq}, tel que $\pi_{q} = f_mw_\K + s$
\end{Def}

\begin{Prop}\label{formeIdeauxBaseFq}
	
	Avec les notation précédentes, et celles de la proposition \ref{baseOrdreWk},
	\begin{equation*}
		\OR_{\Delta} = \left[ 1, fw_\K\right] = \left[ 1, f\frac{\pi_{q} - s}{f_{m}}\right],\; x\in\Z
	\end{equation*}
	De plus, pour tout idéal $\LR\in\OR_{\Delta}$ il existe trois entier entiers $m$, $a$, $b$ et $c$ tels que 
	\begin{equation*}
		\LR = m\left[ a , \frac{-b + \sqrt{\Delta}}{2}\right] = m\left[ a, \frac{c + f\pi_{q}}{f_{m}}\right] 
	\end{equation*}
	
\end{Prop}

\begin{proof}
	Ces résultats découlent de l'égalité $w_{\K} = \frac{\pi_{q} - s}{f_{m}}$ et des propositions \ref{baseOrdreWk},  \ref{formestandard}. En particulier,
	\begin{equation*}
		c = f_m\frac{-b - \sqrt{|\Delta|}}{2}  - sf
	\end{equation*}
	est un entier car $b$ et $\sqrt{|\Delta|}$ ont même parité. 
\end{proof}

On sait évaluer sur toutes courbes elliptiques les endomorphismes entiers et l'endomorphisme de Frobenius. Cependant on ne pourra évaluer un endomorphisme quelconque que sur des points d'ordre premier avec $f_m$.

\begin{Prop}\label{evalEndoAvecDenom}
	Soit $E\in Ell_{\F_{q}}(\OR_{\Delta})$ de conducteur $f$ et conducteur maximal $f_m$. Soit $n\in\N$ premier avec $f_m$, $k\in\N$ un inverse de $f_m$ modulo $n$. Alors pour tout $x\in\Z$,
	\begin{equation*}
		\forall\;P\in E[n],\; \widetilde{\left[ \frac{x + f\pi_{q}}{f_m}\right]}(P) = \left[ k\right] (\left[ x\right]  + f\pi_{q})(P).
	\end{equation*}
\end{Prop}

\begin{proof}
	$\OR_{\Delta}$ est un ordre quadratique imaginaire d'un corps $\K$. Soit $\Lambda$ un réseau associé à un relèvement de Deuring de $E$. On pose pour tout $\lambda\in\K$ le morphisme de tore
	\begin{equation*}
		\phi_{\lambda} : \C/\Lambda\rightarrow \C/\left( \lambda^{-1}\Lambda\right) , z \mapsto z
	\end{equation*}
	Soit $\alpha = \frac{x + f\pi_{q}}{f_m}$. Alors
	\begin{equation*}
		\phi_{\alpha} = [x + f\pi_{q}]\circ\phi_{(f_{m}^{-1})}
	\end{equation*}
	 La restriction de l'endomorphisme $[f_{m}]$ à $E[n]$ est bijective d'inverse $[k]$. Or au niveau des tores $[f_{m}]\circ\phi_{(f_{m}^{-1})}$ est l'identité sur $\C/\Lambda$. Donc après réduction, $\widetilde{\phi_{(f_{m}^{-1})}}$ est l'inverse de $[f_{m}]$ sur $E[n]$. Donc sur $E[n]$, $\phi_{(f_{m}^{-1})} = [k]$ et $\widetilde{\left[ \alpha\right] } = \left[ k\right] (\left[ x\right]  + f\pi_{q})  $
\end{proof}

\begin{Rem}\label{endomorphismeFractionnaire}
	Soit $N\in\N^{*}$ et $\alpha =  \frac{x + f\pi_{q}}{Nf_m}\in\K$. Si $N > 1$, $\alpha$ n'est pas un élément de $\OR_{\Delta}$. Cependant l'idéal fractionnaire principal qu'il engendre agit sur $Ell_{\F_{q}}(\OR_{\Delta})$ comme la classe neutre, donc comme un endomorphisme. Donc un idéal fractionnaire principal défini aussi un endomorphisme, mais dont on ne peut pas expliciter le noyau avec un groupe de torsion. Connaissant $\alpha$, on pourra évaluer l'endomorphisme associé uniquement sur des points d'ordre premier avec le dénominateur $Nf_m$, avec la même méthode que la proposition précédente. 
\end{Rem}





\subsection{Structure du graphe d'isogénie}\label{SECTION volcanIsogenie}

Dans cette section, $E$ est une courbe elliptique ordinaire définie sur $\F_{q}$ de caractéristique $p$. Dans sa thèse \cite{Kohel}, David Kohel décrit la structure des graphes de $l$-isogénies, avec $l$ premier, $l\neq p$. Ce sont des graphes où les sommets sont des courbes elliptiques et où les arêtes représentent les isogénies entre elles. On énonce, en les admettant, les résultats sur ces structures dites de "volcans d'isogénies". On distingue d'abord deux types d'isogénies : celles "horizontales" et celles "verticales".

\begin{Def}
	Soit $E$, $E'$ deux courbes elliptiques ordinaires définie sur $\F_{q}$, d'anneaux d'endomorphismes $\OR$ et $\OR'$. Une isogénie séparable $\phi : E\rightarrow E'$ est dite horizontale si $\OR = \OR'$, verticale sinon.
\end{Def}

\begin{Prop}[\cite{Kohel}(Prop 21)]
	
	Soit $E$, $E'$ deux courbes elliptiques ordinaire définie sur $\F_{q}$, d'anneaux d'endomorphismes $\OR$ et $\OR'$. Soit $l$ un nombre premier différent de $p$ et $\phi : E\rightarrow E'$ une $l$-isogénie séparable. Alors
	\begin{enumerate}
		\item Soit $\OR = \OR'$ et $\phi$ est horizontale
		\item Soit $\OR'\subset \OR$, $\left[ \OR : \OR' \right] = l$ et $\phi$ est verticale "descendante"
		\item Soit $\OR\subset \OR'$, $\left[ \OR' : \OR \right] = l$ et $\phi$ est verticale "ascendante" 
	\end{enumerate}
\end{Prop}

\begin{Rem}
	Une $l$-isogénie est déterminée par son noyau, qui est l'un des sous groupes d'ordre $l$ de $E$. On a vu à la proposition \ref{NbDeLIsogenies} que cela implique l'existence d'exactement $l+1$ $l$-isogénies sur $E$ définies sur $\Kb$. 
\end{Rem}

On s'intéresse maintenant aux $l$-isogénies définies sur $\F_{q}$.

\begin{Prop}[\cite{Kohel}(Prop 21, 22, 23)]\label{KohelStructureVolcan}
	
	Soit $E$ une courbe elliptique à multiplication complexe définie sur $\F_{q}$, de conducteur $f$ et de conducteur maximal $f_{m}$. Soit
	$r = f_{m}/f$. Alors
	\begin{enumerate}
		\item Si $l$ ne divise pas $f$, il existe exactement $1 + \left( \frac{\Delta}{l}\right)$ $l$-isogénies horizontales définies sur $\F_{q}$, et aucune ascendante.
		\item Si $l$ divise $f$, il existe une unique $l$-isogénie ascendante définie sur $\F_{q}$, et aucune horizontale.
	\end{enumerate}
	De plus il peut exister des isogénies supplémentaires descendantes définie sur $\F_{q}$ selon si $l$ divise $r$ 
	\begin{enumerate}
		\item Si $l$ ne divise pas $r$, il n'existe aucune $l$-isogénie supplémentaire définie sur $\F_{q}$
		\item Si $l$ divise $r$, toutes les $l+1$ $l$-isogénies sont définies sur $\F_{q}$
	\end{enumerate}
	
\end{Prop}

\begin{Rem}
	Ainsi "$l$ divise $f$" est une "condition d'existence d'une ascension", tandis que "$l$ ne divise pas $f$" est une "condition d'existence d'horizontalités". Ces critères ne dépendent pas du corps $\F_{q}$ sur lequel on considère $E$. En plus de cela, "$l$ divise $f_m/f$" est une "condition d'existence de descentes", qui cette fois dépend du corps de base considéré. Si celui-ci est une extension de degré suffisamment grand, les isogénies descendantes seront toutes définies et le conducteur maximal associé $f_m$ sera tel que $l$ divise $f_m / f$. 
\end{Rem}

\begin{Rem}
	Puisqu'il existe au plus une $l$-isogénie ascendante, toutes les isogénies descendantes ont pour images des courbes elliptiques distinctes. Partant du "cratère" du graphe, c'est à dire $Ell(\OR_{\K})$, on a donc de plus en plus de courbes elliptiques à chaque "niveau" $Ell(\OR_{\Delta})$, jusque au niveau minimal $Ell(\Z\left[ \pi_{q}\right] )$. D'où le nom de "volcan de $l$-isogénies".
\end{Rem}



\begin{Rem}
	Pour tout nombre premier $l$ ne divisant pas $f_{m}$, et $l\neq p$, il existe $1 + \left( \frac{\Delta}{l}\right)$ $l$-isogénies horizontales, c'est à dire $0$, $1$ ou $2$, et aucune ascendante ni descendante. On peut faire le lien avec la ramification de $l$ dans $\OR_{\Delta}$. En effet $l$ est ici premier avec $f$, donc les idéaux de normes $l$ sont premier avec le conducteur, donc propre. Ainsi on peut calculer des groupes de torsions à partir d'idéaux de norme $l$ et utiliser la proposition \ref{DefActionIdeauxSurCourbeOrdinaire} pour en déduire des noyaux de $l$-isogénies rationnelles. 
	\begin{enumerate}
		\item Si $l$ est décomposé, $\left( \frac{\Delta}{l}\right) = 1$ et il existe $2$ $l$-isogénies rationnelles, qui sont horizontales. On peut expliciter leurs noyaux, ce sont les deux groupes de torsions des deux idéaux premiers factorisant $l\OR_{\Delta}$.
		
		\item Si $l$ est ramifié, $\left( \frac{\Delta}{l}\right) = 0$ et il existe $1$ $l$-isogénie rationnelle, horizontale. Son noyau est le groupe de torsion de l'idéal premier dont le carré est $l\OR_{\Delta}$.
		
		\item Si $l$ est inerte, $\left( \frac{\Delta}{l}\right) = -1$. Il n'existe pas de $l$-isogénie rationnelle, tout comme il n'existe pas d'idéaux premier de norme $l$.
	\end{enumerate}

	Ainsi dans le cas où $l$ ne divise pas $f_m$, on peut dénombrer et représenter toutes les $l$-isogénies entre courbes de $Ell_{\F_{q}}(\OR_{\Delta})$ par des idéaux d'ordres quadratiques imaginaires, et cela pour tout ordre de conducteur $f$ divisant $f_m$.   
\end{Rem}
