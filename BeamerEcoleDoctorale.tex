		\begin{itemize}
			\item Problème DLP sur le cercle : Facile.
			\item Ellipses, paraboles, hyperboles... : courbe de "genre" $0$ donc "simple".
		\begin{block}{Courbes Elliptiques}<2->
			Les courbes elliptiques sont les courbes de genre $1$.
		\end{block}
		
		\item<2-> Problème DLP sur courbes elliptiques : Difficile.

	\begin{frame} 
		\frametitle{Cryptanalyse et isogénies} 
		
		Si $E$ et $E'$ sont deux courbes elliptiques, soit $\phi$ une application qui à un point $P\in E$ associe un point $\phi(P)\in E'$.
		
		\begin{block}{Isogénie entre courbes elliptiques}<2->
			$\phi$ est une isogénie si elle vérifie
			\begin{equation*}
				\phi(P+Q) = \phi(P) + \phi(Q).
			\end{equation*}
		\end{block}
		
		\begin{block}{Lien avec le problème DLP }<3->
			Si $\phi$ est une isogénie, alors $Q = k\cdot P\iff \phi(Q) = k\cdot\phi(P)$
		\end{block}
		
	\end{frame}
	
	\begin{frame}
		\frametitle{Calculer une isogénie}
		\begin{block}{Sujet de stage au laboratoire MIS}
			Trouver $\phi$ isogénie entre $E$ et $E'$ ? Calculer $\phi(P)$ lorsque $P\in E$?
		\end{block}
		\begin{itemize}
			\item<2-> Complexité exprimée selon $q$ (\textit{taille} des groupes) et $l$ (\text{taille} de l'isogénie).
			\item<2-> Formule de Vélu : méthode générale, polynomiale en $\log(q)$ mais exponentielle $\log(l)$ : Permet de calculer les \textit{petites} isogénies.
		\end{itemize}
		
		\begin{block}{Méthodes de calcul d'isogénies}<3->
			\begin{itemize}
				\item 2008 : \textbf{Action du groupe de classes}: polynomiale en $\log(l)$, mais sous-exponentiel en $q$. 
				\item 2023 : \textbf{Clapoti}: polynomial selon $l$ et $q$. 
			\end{itemize}
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Action du groupe de classes d'idéaux}
		\begin{block}{Calcul d'une isogénie}
			\begin{enumerate}
				\item On représente une isogénie par un idéal $L$ de classe $\tilde{L}$.
				\item<2-> $\tilde{L}$ s'écrit dans le groupe de classes $\tilde{L} = \tilde{L_1}\cdots\tilde{L_n}$.
				\item<3-> On calcule les isogénies associées à $L_1, \cdots, L_n$ avec Vélu.
				\item<4-> On en déduit l'isogénie $\phi$.
			\end{enumerate}
		\end{block}
		\begin{itemize}
			\item<5-> $L_1,\cdots,L_n$ : \textit{petites} isogénies, calculées avec Vélu.
			\item<5-> Trouver $\tilde{L} = \tilde{L_1}\cdots\tilde{L_n}$ est exponentielle $\log(q)$ : Optimisation possible. 
		\end{itemize}
		
		\begin{block}{Contribution personnelle}<6->
			\begin{itemize}
				\item Proposition d'optimisation pour l'étape $2$, implémentée dans Sage. 
				\item Comparaison avec l'état de l'art (Jao Soukharev 2010).
			\end{itemize}
		
		\end{block}
	
	\end{frame}
