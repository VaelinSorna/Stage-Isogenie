	\begin{frame} 
		\frametitle{Cryptanalyse et isogénies} 
		
		\begin{itemize}
			\item Si $E$ et $E'$ sont deux courbes elliptiques, soit $\phi : E\rightarrow E'$.
		\end{itemize}
			
		\begin{block}{Isogénie entre courbes elliptiques}<2->
			Pour $P$, $Q\in E$, $\phi$ vérifie
			\begin{equation*}
				\phi(P+Q) = \phi(P) + \phi(Q).
			\end{equation*}
		\end{block}
		
		\begin{block}{Lien avec le problème DLP }<3->
			Si $P$, $Q\in E$, alors $Q = k\cdot P\iff \phi(Q) = k\cdot\phi(P)$
		\end{block}
		
		\begin{itemize}
			\item<3-> Si DLP plus simple sur $E'$, $\phi$ permet d'attaquer DLP sur $E$.
		\end{itemize}
		
	\end{frame}
	
	\begin{frame}
		\frametitle{Calculer une isogénie}
		\begin{block}{Sujet de stage au laboratoire MIS}
			Trouver $\phi$ isogénie entre $E$ et $E'$ ? Calculer $\phi(P)$ lorsque $P\in E$?
		\end{block}
		\begin{itemize}
			\item<2-> Complexité selon $q$ (taille des groupes) et $l$ (\textit{taille} de l'isogénie).
			\item<2-> Pour les \textit{petites} isogénies: Formule de Vélu, polynomiale en $\log(q)$ mais exponentielle en $\log(l)$. 
		\end{itemize}
		
		\begin{block}{Méthodes de calculs d'isogénies}<3->
			\begin{itemize}
				\item 2008 : \textbf{Action du groupe de classes d'idéaux}: polynomiale en $\log(l)$, mais sous-exponentielle en $q$. 
				\item 2023 : \textbf{Clapoti}: polynomiale selon $l$ et $q$ pour certaines familles de courbes. 
			\end{itemize}
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Action du groupe de classes d'idéaux}
		\begin{block}{Calcul d'une isogénie}
			\begin{enumerate}
				\item On représente une isogénie par un idéal $L$ de classe $\tilde{L}$.
				\item<2-> $\tilde{L}$ s'écrit dans le groupe de classes $\tilde{L} = \tilde{L_1}\cdot\ldots\cdot\tilde{L_n}$.
				\item<3-> On calcule les isogénies associées à $L_1, \ldots, L_n$.
				\item<4-> On en déduit l'isogénie $\phi$.
			\end{enumerate}
		\end{block}
		\begin{itemize}
			\item<4-> $L_1,\ldots,L_n$ : \textit{petites} isogénies, calculées avec Vélu. 
		\end{itemize}
		
		\begin{block}{Contribution personnelle}<5->
			\begin{itemize}
				\item Proposition d'optimisation pour l'étape $2$, implémentée dans Sage. 
				\item Comparaison avec l'état de l'art (Jao Soukharev 2010).
			\end{itemize}
		
		\end{block}
	
	\end{frame}
